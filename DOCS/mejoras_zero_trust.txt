+fn usage() -> ExitCode {
+    eprintln!("uso: livencrypt-boot-shield <seal|unseal|verify> <clave-hex-32B> <entrada> [salida]");
+    ExitCode::from(2)

+fn main() -> ExitCode {
+    let args: Vec<String> = std::env::args().collect();
+    if args.len() < 4 {
+        return usage();
+    }
+
+    let command = args[1].as_str();
+    let key = match BootKey::from_hex(&args[2]) {
+        Ok(key) => key,
+        Err(error) => {
+            eprintln!("error: {}", error);
+            return ExitCode::FAILURE;
+        }
+    };
+    let input = match std::fs::read(&args[3]) {
+        Ok(bytes) => bytes,
+        Err(error) => {
+            eprintln!("error leyendo {}: {}", args[3], error);
+            return ExitCode::FAILURE;
+        }
+    };
+
+    match command {
+        "seal" => write_result(boot_shield::seal_config(&key, &input), args.get(4)),
+        "unseal" => write_result(boot_shield::open_config(&key, &input), args.get(4)),
+        "verify" => {
+            if boot_shield::verify_config(&key, &input) {
+                println!("ok");
+                ExitCode::SUCCESS
+            } else {
+                eprintln!("verificacion fallida");
+                ExitCode::FAILURE
+            }
+        }
+        _ => usage(),
+    }
+}

+fn write_result(result: Result<Vec<u8>, String>, output: Option<&String>) -> ExitCode {
+    let Some(output) = output else {
+        return usage();
+    };

+        Ok(bytes) => match std::fs::write(output, bytes) {
+            Ok(()) => ExitCode::SUCCESS,
+            Err(error) => {
+                eprintln!("error escribiendo {}: {}", output, error);
+                ExitCode::FAILURE
+            }
+        },
+        Err(error) => {
+            eprintln!("error: {}", error);
+            ExitCode::FAILURE
+        }
+    }

+fn argon2id() -> Result<Argon2<'static>, String> {

+fn derive_key(secret: &str, salt: &[u8]) -> Result<[u8; 32], String> {
+    if secret.trim().chars().count() < MIN_SECRET_CHARS {
+        return Err(format!(
+            "la clave de texto debe tener al menos {} caracteres",
+            MIN_SECRET_CHARS
+        ));
+    }
+    let mut key = [0u8; 32];
+    argon2id()?
+        .hash_password_into(secret.as_bytes(), salt, &mut key)
+        .map_err(|error| format!("no se pudo derivar la clave de texto: {}", error))?;
+    Ok(key)
+}

+pub fn encrypt_text(plaintext: &str, secret: &str) -> Result<String, String> {
+    let mut salt = [0u8; SALT_LEN];
+    rand::rngs::OsRng.fill_bytes(&mut salt);
+    let mut nonce = [0u8; NONCE_LEN];
+    rand::rngs::OsRng.fill_bytes(&mut nonce);
+
+    let mut header = Vec::with_capacity(MAGIC.len() + SALT_LEN + NONCE_LEN);
+    header.extend_from_slice(MAGIC);
+    header.extend_from_slice(&salt);
+    header.extend_from_slice(&nonce);
+
+    let mut key = derive_key(secret, &salt)?;
+    let cipher = XChaCha20Poly1305::new(Key::from_slice(&key));
+    let ciphertext = cipher.encrypt(
+        XNonce::from_slice(&nonce),
+        Payload {
+            msg: plaintext.as_bytes(),
+            aad: &header,
+        },
+    );
+    key.zeroize();
+    let ciphertext = ciphertext.map_err(|_| "fallo AEAD al cifrar texto".to_string())?;
+
+    let mut envelope = header;
+    envelope.extend_from_slice(&ciphertext);
+    Ok(hex::encode(envelope))
+}

+    if raw.len() <= header_len || &raw[..MAGIC.len()] != MAGIC {
+        return Err("texto cifrado Livencrypt no reconocido".to_string());
+    }
+    let salt = &raw[MAGIC.len()..MAGIC.len() + SALT_LEN];
+    let nonce = &raw[MAGIC.len() + SALT_LEN..header_len];

+    let cipher = XChaCha20Poly1305::new(Key::from_slice(&key));
+    let plaintext = cipher.decrypt(
+        XNonce::from_slice(nonce),
+        Payload {
+            msg: ciphertext,
+            aad: &raw[..header_len],
+        },
+    );
