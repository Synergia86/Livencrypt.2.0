#!/bin/sh
set -eu

ROOT_DIR="$(CDPATH= cd -- "$(dirname -- "$0")/.." && pwd)"
VERSION="$(sed -n 's/^version = "\(.*\)"/\1/p' "$ROOT_DIR/Cargo.toml" | head -n 1)"
TARGET_DIR="${CARGO_TARGET_DIR:-$ROOT_DIR/target}"
PACKAGE_NAME="livencrypt-core"
DEBIAN_WORK_DIR="$TARGET_DIR/debian"

detect_deb_arch() {
  if command -v dpkg >/dev/null 2>&1; then
    dpkg --print-architecture
    return
  fi

  case "$(uname -m)" in
    x86_64) echo "amd64" ;;
    aarch64|arm64) echo "arm64" ;;
    armv7l|armv7*) echo "armhf" ;;
    i386|i686) echo "i386" ;;
    *)
      echo "amd64" >&2
      echo "Livencrypt: arquitectura no reconocida; usando amd64 por defecto." >&2
      ;;
  esac
}

ARCH="$(detect_deb_arch)"
BUILD_ROOT="$TARGET_DIR/debian/${PACKAGE_NAME}_${VERSION}_${ARCH}"
DEB_PATH="$TARGET_DIR/debian/${PACKAGE_NAME}_${VERSION}_${ARCH}.deb"

CARGO_TARGET_DIR="$TARGET_DIR" cargo build --release --locked --manifest-path "$ROOT_DIR/Cargo.toml"

rm -rf "$BUILD_ROOT"
mkdir -p "$BUILD_ROOT/DEBIAN"
mkdir -p "$BUILD_ROOT/usr/bin"
mkdir -p "$BUILD_ROOT/usr/share/doc/$PACKAGE_NAME"
mkdir -p "$BUILD_ROOT/usr/share/applications"
mkdir -p "$BUILD_ROOT/usr/share/icons/hicolor/128x128/apps"
mkdir -p "$BUILD_ROOT/usr/share/livencrypt/apparmor"
mkdir -p "$BUILD_ROOT/usr/share/livencrypt/native-messaging"
mkdir -p "$BUILD_ROOT/usr/share/livencrypt/scripts"

install -m 0755 "$TARGET_DIR/release/livencrypt-core" "$BUILD_ROOT/usr/bin/livencrypt-core"
install -m 0755 "$TARGET_DIR/release/livencrypt-bridge" "$BUILD_ROOT/usr/bin/livencrypt-bridge"
install -m 0755 "$ROOT_DIR/scripts/livencrypt-core-window.sh" "$BUILD_ROOT/usr/bin/livencrypt-core-window"
install -m 0755 "$ROOT_DIR/scripts/postinstall.sh" "$BUILD_ROOT/DEBIAN/postinst"
install -m 0755 "$ROOT_DIR/scripts/postremove.sh" "$BUILD_ROOT/DEBIAN/postrm"
install -m 0644 "$ROOT_DIR/packaging/desktop/livencrypt-core.desktop" "$BUILD_ROOT/usr/share/applications/livencrypt-core.desktop"
install -m 0644 "$ROOT_DIR/icons/livencrypt2.0_logo.png" "$BUILD_ROOT/usr/share/icons/hicolor/128x128/apps/livencrypt-core.png"
install -m 0644 "$ROOT_DIR/packaging/apparmor/livencrypt-core" "$BUILD_ROOT/usr/share/livencrypt/apparmor/livencrypt-core"
install -m 0644 "$ROOT_DIR/scripts/com.livencrypt.gateway.json" "$BUILD_ROOT/usr/share/livencrypt/native-messaging/com.livencrypt.gateway.json"
install -m 0755 "$ROOT_DIR/scripts/initramfs-build.sh" "$BUILD_ROOT/usr/share/livencrypt/scripts/initramfs-build.sh"
install -m 0755 "$ROOT_DIR/scripts/initramfs-install.sh" "$BUILD_ROOT/usr/share/livencrypt/scripts/initramfs-install.sh"

cat > "$BUILD_ROOT/DEBIAN/control" <<EOF
Package: $PACKAGE_NAME
Version: $VERSION
Section: utils
Priority: optional
Architecture: $ARCH
Maintainer: Synergia86
Depends: libc6, libcap2-bin
Description: Livencrypt 2.0 - TUI segura para texto, archivo y plantillas
 Interfaz de terminal para Livencrypt 2.0 sin WebView ni frontend web.
 Incluye hardening de proceso, Boot Shield y nucleo criptografico local.
EOF

cat > "$BUILD_ROOT/usr/share/doc/$PACKAGE_NAME/README.Debian" <<EOF
Livencrypt 2.0 TUI
==================

Ejecute desde terminal:

  livencrypt-core

Diagnostico de hardening:

  livencrypt-core --hardening-check
EOF

if command -v dpkg-deb >/dev/null 2>&1; then
  dpkg-deb --build "$BUILD_ROOT" "$DEB_PATH"
else
  command -v ar >/dev/null 2>&1 || {
    echo "Livencrypt: falta 'dpkg-deb' o 'ar' para generar el paquete .deb." >&2
    echo "Instale dpkg, dpkg-dev o binutils segun su distribucion." >&2
    exit 1
  }
  command -v tar >/dev/null 2>&1 || {
    echo "Livencrypt: falta 'tar' para generar el paquete .deb." >&2
    exit 1
  }

  PKG_WORK="$DEBIAN_WORK_DIR/.${PACKAGE_NAME}_${VERSION}_${ARCH}.deb-work"
  rm -rf "$PKG_WORK"
  mkdir -p "$PKG_WORK"

  printf "2.0\n" > "$PKG_WORK/debian-binary"
  (cd "$BUILD_ROOT/DEBIAN" && tar --sort=name --owner=0 --group=0 --numeric-owner -czf "$PKG_WORK/control.tar.gz" .)
  (cd "$BUILD_ROOT" && tar --sort=name --owner=0 --group=0 --numeric-owner --exclude=./DEBIAN -czf "$PKG_WORK/data.tar.gz" .)

  rm -f "$DEB_PATH"
  (cd "$PKG_WORK" && ar rcs "$DEB_PATH" debian-binary control.tar.gz data.tar.gz)
  rm -rf "$PKG_WORK"
fi
echo "$DEB_PATH"

#!/bin/sh
set -eu

ROOT_DIR="$(CDPATH= cd -- "$(dirname -- "$0")/.." && pwd)"
TARGET="${LIVENCRYPT_INITRAMFS_TARGET:-x86_64-unknown-linux-musl}"
OUT_DIR="$ROOT_DIR/target/initramfs"

command -v rustup >/dev/null 2>&1 || {
  echo "Livencrypt PBA: rustup no esta disponible." >&2
  exit 1
}

rustup target add "$TARGET"
cargo build --release --locked --target "$TARGET" --manifest-path "$ROOT_DIR/Cargo.toml"

install -d -m 0755 "$OUT_DIR"
install -m 0755 \
  "$ROOT_DIR/target/$TARGET/release/livencrypt-core" \
  "$OUT_DIR/livencrypt-core"

cat > "$OUT_DIR/livencrypt_hook" <<EOF
#!/bin/sh
PREREQ=""

prereqs() {
  echo "\$PREREQ"
}

case "\$1" in
  prereqs)
    prereqs
    exit 0
    ;;
esac

. /usr/share/initramfs-tools/hook-functions
copy_exec /usr/bin/livencrypt-core /bin/livencrypt-core
EOF

chmod 0755 "$OUT_DIR/livencrypt_hook"

echo "Livencrypt PBA: artefactos generados en $OUT_DIR."
echo "Livencrypt PBA: no se ha instalado ningun hook de arranque."

#!/bin/sh
set -eu

ROOT_DIR="$(CDPATH= cd -- "$(dirname -- "$0")/.." && pwd)"
HOOK_SRC="$ROOT_DIR/target/initramfs/livencrypt_hook"
HOOK_DST="/etc/initramfs-tools/hooks/livencrypt-core"
BACKUP_DIR="/var/backups/livencrypt-initramfs"

if [ "$(id -u)" -ne 0 ]; then
  echo "Livencrypt PBA: ejecute este script como root." >&2
  exit 1
fi

if [ "${LIVENCRYPT_ENABLE_PBA:-0}" != "1" ]; then
  echo "Livencrypt PBA: instalacion bloqueada." >&2
  echo "Defina LIVENCRYPT_ENABLE_PBA=1 solo tras backup y prueba en equipo no critico." >&2
  exit 1
fi

if [ ! -f "$HOOK_SRC" ]; then
  echo "Livencrypt PBA: falta $HOOK_SRC. Ejecute scripts/initramfs-build.sh primero." >&2
  exit 1
fi

command -v update-initramfs >/dev/null 2>&1 || {
  echo "Livencrypt PBA: update-initramfs no esta disponible." >&2
  exit 1
}

install -d -m 0700 "$BACKUP_DIR"
if [ -d /boot ]; then
  tar -C /boot -czf "$BACKUP_DIR/boot-initramfs-$(date +%Y%m%d%H%M%S).tar.gz" .
fi

install -m 0755 "$HOOK_SRC" "$HOOK_DST"
update-initramfs -u

echo "Livencrypt PBA: hook instalado en $HOOK_DST."
echo "Livencrypt PBA: backup creado en $BACKUP_DIR."

#!/bin/sh
set -eu

TITLE="Livencrypt 2.0"
CMD="/usr/bin/livencrypt-core"
WM_CLASS="LivencryptCore"
ICON="/usr/share/icons/hicolor/128x128/apps/livencrypt-core.png"

if command -v xfce4-terminal >/dev/null 2>&1; then
  exec xfce4-terminal --class="$WM_CLASS" --icon="$ICON" --title="$TITLE" --command="$CMD"
fi

if command -v mate-terminal >/dev/null 2>&1; then
  exec mate-terminal --class="$WM_CLASS" --title="$TITLE" -e "$CMD"
fi

if command -v gnome-terminal >/dev/null 2>&1; then
  exec gnome-terminal --class="$WM_CLASS" --title="$TITLE" -- "$CMD"
fi

if command -v konsole >/dev/null 2>&1; then
  exec konsole --new-tab --workdir "$HOME" --qwindowtitle "$TITLE" -p tabtitle="$TITLE" -e "$CMD"
fi

if command -v xterm >/dev/null 2>&1; then
  exec xterm -class "$WM_CLASS" -name "$WM_CLASS" -T "$TITLE" -e "$CMD"
fi

if command -v x-terminal-emulator >/dev/null 2>&1; then
  exec x-terminal-emulator -T "$TITLE" -e "$CMD"
fi

exec "$CMD"

#!/bin/sh
set -eu

if [ "${1:-}" != "configure" ]; then
  exit 0
fi

binary_path="/usr/bin/livencrypt-core"
integrity_dir="/etc/livencrypt"
integrity_file="$integrity_dir/expected.sha256"
install_profile="${LIVENCRYPT_INSTALL_PROFILE:-expu}"
apparmor_src="/usr/share/livencrypt/apparmor/livencrypt-core"
apparmor_dst="/etc/apparmor.d/usr.bin.livencrypt-core"
native_manifest_src="/usr/share/livencrypt/native-messaging/com.livencrypt.gateway.json"
native_manifest_name="com.livencrypt.gateway.json"

detect_os() {
  if [ -f /etc/os-release ]; then
    # shellcheck disable=SC1091
    . /etc/os-release
    echo "${ID:-unknown}:${ID_LIKE:-}"
  else
    uname -s
  fi
}

configure_memlock_limits() {
  if [ -d /etc/security/limits.d ]; then
    cat > /etc/security/limits.d/99-livencrypt.conf <<EOF
* soft memlock unlimited
* hard memlock unlimited
root soft memlock unlimited
root hard memlock unlimited
EOF
    chmod 0644 /etc/security/limits.d/99-livencrypt.conf
  fi
}

configure_expected_hash() {
  if ! command -v sha256sum >/dev/null 2>&1; then
    echo "Livencrypt: sha256sum no esta disponible; no se puede registrar hash de integridad." >&2
    exit 1
  fi

  install -d -m 0755 "$integrity_dir"
  sha256sum "$binary_path" | awk '{print $1}' > "$integrity_file"
  chmod 0644 "$integrity_file"
}

configure_profile() {
  install -d -m 0755 "$integrity_dir"
  case "$install_profile" in
    standard|isolated|expu|pba) ;;
    *)
      echo "Livencrypt: perfil '$install_profile' no valido; usando expu." >&2
      install_profile="expu"
      ;;
  esac
  printf "%s\n" "$install_profile" > "$integrity_dir/install_profile.conf"
  chmod 0644 "$integrity_dir/install_profile.conf"
}

configure_sandbox_profile() {
  install -d -m 0755 "$integrity_dir/sandbox"
  cat > "$integrity_dir/sandbox/profile.conf" <<EOF
network=isolated
mount=private
cpu_seconds=120
memory_mb=512
file_size_mb=64
EOF
  chmod 0644 "$integrity_dir/sandbox/profile.conf"
}

configure_apparmor_profile() {
  if [ ! -f "$apparmor_src" ]; then
    return 0
  fi

  if ! command -v apparmor_parser >/dev/null 2>&1 || [ ! -d /etc/apparmor.d ]; then
    echo "Livencrypt: AppArmor no disponible; perfil expu no cargado." >&2
    return 0
  fi

  install -m 0644 "$apparmor_src" "$apparmor_dst"
  if ! apparmor_parser -r "$apparmor_dst"; then
    echo "Livencrypt: aviso, no se pudo cargar el perfil AppArmor." >&2
  fi
}

register_desktop_entry() {
  if command -v update-desktop-database >/dev/null 2>&1 && [ -d /usr/share/applications ]; then
    update-desktop-database /usr/share/applications >/dev/null 2>&1 || true
  fi
}

register_native_messaging_host() {
  if [ ! -f "$native_manifest_src" ]; then
    echo "Livencrypt: manifiesto de mensajeria nativa no encontrado en $native_manifest_src." >&2
    return 0
  fi

  for manifest_dir in \
    /etc/opt/chrome/native-messaging-hosts \
    /etc/chromium/native-messaging-hosts \
    /etc/brave/native-messaging-hosts \
    /etc/opt/brave.com/brave/native-messaging-hosts
  do
    install -d -m 0755 "$manifest_dir"
    install -m 0644 "$native_manifest_src" "$manifest_dir/$native_manifest_name"
  done
}

if [ ! -x "$binary_path" ]; then
  echo "Livencrypt: no se encontro el binario instalado en $binary_path." >&2
  exit 1
fi

os_id="$(detect_os)"
configure_profile

if ! command -v setcap >/dev/null 2>&1; then
  echo "Livencrypt: setcap no esta disponible. Instale libcap2-bin para habilitar CAP_IPC_LOCK." >&2
  exit 1
fi

setcap cap_ipc_lock=ep "$binary_path" || {
  echo "Livencrypt: ERROR no se pudieron asignar capacidades al binario." >&2
  echo "Livencrypt: verifique que el sistema de archivos permita setcap y no este montado con nosuid." >&2
  exit 1
}

if command -v getcap >/dev/null 2>&1; then
  if ! getcap "$binary_path" | grep -q "cap_ipc_lock"; then
    echo "Livencrypt: CRITICO CAP_IPC_LOCK no quedo aplicado sobre $binary_path." >&2
    echo "Livencrypt: mlockall real puede fallar sin esta capacidad." >&2
    exit 1
  fi
fi

memlock_limit="$(ulimit -l 2>/dev/null || echo 0)"
case "$memlock_limit" in
  unlimited) ;;
  ''|*[!0-9]*)
    echo "Livencrypt: AVISO no se pudo interpretar el limite memlock: $memlock_limit" >&2
    ;;
  *)
    if [ "$memlock_limit" -lt 65536 ]; then
      echo "Livencrypt: AVISO el limite memlock es bajo (${memlock_limit} KB)." >&2
      echo "Livencrypt: considere elevarlo en /etc/security/limits.conf si MLOCKALL aparece como FALLO." >&2
    fi
    ;;
esac

case "$os_id" in
  *debian*|*ubuntu*|*linuxmint*|debian:*|ubuntu:*|linuxmint:*)
    configure_memlock_limits
    configure_sandbox_profile
    ;;
  *fedora*|*rhel*|*centos*)
    configure_sandbox_profile
    echo "Livencrypt: SELinux detectado o probable; politica especifica pendiente." >&2
    ;;
  *)
    configure_sandbox_profile
    echo "Livencrypt: SO detectado '$os_id'; aplicando hardening Linux disponible." >&2
    ;;
esac

configure_expected_hash
register_desktop_entry
register_native_messaging_host

echo "Livencrypt: CAP_IPC_LOCK aplicado a $binary_path para mlockall real."
echo "Livencrypt: SHA-256 esperado registrado en $integrity_file."
echo "Livencrypt: perfil expu instalado: $install_profile."
echo "Livencrypt: perfil sandbox local configurado en $integrity_dir/sandbox/profile.conf."
echo "Livencrypt: host de mensajeria nativa registrado como /usr/bin/livencrypt-bridge."
echo "Livencrypt: AppArmor queda disponible, pero NO se activa automaticamente. Activelo desde Configuracion."
if [ "$install_profile" = "pba" ]; then
  echo "Livencrypt: PBA seleccionado, pero initramfs NO se activa automaticamente. Use script dedicado y rollback." >&2
fi
exit 0

#!/bin/sh
set -eu

if [ "${1:-}" != "purge" ]; then
  exit 0
fi

rm -f /etc/livencrypt/expected.sha256
rm -f /etc/livencrypt/install_profile.conf
rm -f /etc/livencrypt/sandbox/profile.conf
rmdir /etc/livencrypt/sandbox 2>/dev/null || true
rmdir /etc/livencrypt 2>/dev/null || true
rm -f /etc/security/limits.d/99-livencrypt.conf
rm -f /etc/opt/chrome/native-messaging-hosts/com.livencrypt.gateway.json
rm -f /etc/chromium/native-messaging-hosts/com.livencrypt.gateway.json
rm -f /etc/brave/native-messaging-hosts/com.livencrypt.gateway.json
rm -f /etc/opt/brave.com/brave/native-messaging-hosts/com.livencrypt.gateway.json

if [ -f /etc/apparmor.d/usr.bin.livencrypt-core ]; then
  if command -v apparmor_parser >/dev/null 2>&1; then
    apparmor_parser -R /etc/apparmor.d/usr.bin.livencrypt-core 2>/dev/null || true
  fi
  rm -f /etc/apparmor.d/usr.bin.livencrypt-core
fi

if command -v update-desktop-database >/dev/null 2>&1 && [ -d /usr/share/applications ]; then
  update-desktop-database /usr/share/applications >/dev/null 2>&1 || true
fi

echo "Livencrypt: purge completado. Los datos de usuario/vaults no se eliminan automaticamente."
echo "Livencrypt: para borrar datos personales, elimine manualmente ~/.local/share/livencrypt tras hacer copia de seguridad."

exit 0
