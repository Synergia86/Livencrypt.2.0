use crate::security::vault::derive_vault_key;
use zeroize::Zeroize;

#[derive(Debug, Clone, PartialEq)]
pub enum GatewaySecurityState {
    Operational,
    Frozen { reason: String, timestamp: u64 },
    HardLocked,
}

pub struct TacticalCircuitBreaker {
    pub state: GatewaySecurityState,
    pub integrity_failures: u32,
}

impl TacticalCircuitBreaker {
    pub fn new() -> Self {
        Self {
            state: GatewaySecurityState::Operational,
            integrity_failures: 0,
        }
    }

    pub fn trip_breaker(&mut self, reason: &str, timestamp: u64) {
        self.state = GatewaySecurityState::Frozen {
            reason: reason.to_string(),
            timestamp,
        };
    }

    pub fn trigger_hard_lock(&mut self) {
        self.state = GatewaySecurityState::HardLocked;
    }

    pub fn resolve_and_reset(
        &mut self,
        cajas: [i64; 5],
        master_phrase: &str,
    ) -> Result<(), String> {
        if self.state == GatewaySecurityState::HardLocked {
            return Err(
                "bloqueo perimetral permanente: reinicio completo del binario requerido"
                    .to_string(),
            );
        }

        let mut verification_key = derive_vault_key(cajas, master_phrase)?;
        if verification_key.is_empty() {
            verification_key.zeroize();
            return Err("credenciales de autorizacion locales invalidas".to_string());
        }

        verification_key.zeroize();
        self.state = GatewaySecurityState::Operational;
        self.integrity_failures = 0;
        Ok(())
    }
}

impl Default for TacticalCircuitBreaker {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct FirewallRules {
    pub max_daily_tx: u32,
    pub max_daily_volume_units: Decimal,
    pub allowed_dex_contracts: Vec<String>,
    pub allowed_cex_destinations: Vec<String>,
    pub allowed_erc20_contracts: Vec<String>,
    pub allowed_chain_ids: Vec<u64>,
    /// Decimales registrados por dirección de contrato ERC-20.
    /// Sin registro en `token_decimals`, la transacción ERC-20 es rechazada.
    pub token_decimals: HashMap<String, u8>,
}

impl Default for FirewallRules {
    fn default() -> Self {
        Self {
            max_daily_tx: 10,
            max_daily_volume_units: Decimal::new(5000, 0),
            allowed_dex_contracts: Vec::new(),
            allowed_cex_destinations: Vec::new(),
            allowed_erc20_contracts: Vec::new(),
            allowed_chain_ids: vec![0, 1, 501],
            token_decimals: HashMap::new(),
        }
    }
}

pub struct LiveFirewall {
    pub rules: FirewallRules,
    pub tx_count_today: u32,
    pub volume_today_units: Decimal,
    pub last_reset_timestamp: u64,
}

impl Default for LiveFirewall {
    fn default() -> Self {
        Self::new(FirewallRules::default())
    }
}

impl LiveFirewall {
    pub fn new(rules: FirewallRules) -> Self {
        let now = now_unix();
        Self {
            rules,
            tx_count_today: 0,
            volume_today_units: Decimal::ZERO,
            last_reset_timestamp: now,
        }
    }

    fn check_and_reset_daily_counters(&mut self) {
        let now = now_unix();
        if now.saturating_sub(self.last_reset_timestamp) >= 86_400 {
            self.tx_count_today = 0;
            self.volume_today_units = Decimal::ZERO;
            self.last_reset_timestamp = now;
        }
    }

    pub fn validate_parsed_transaction(&mut self, parsed: &ParsedTx) -> Result<(), String> {
        self.check_and_reset_daily_counters();

        if self.tx_count_today >= self.rules.max_daily_tx {
            return Err(format!(
                "restriccion de frecuencia: limite maximo de {} transacciones diarias alcanzado",
                self.rules.max_daily_tx
            ));
        }

        // Anti-replay: el chain_id parseado debe estar permitido para evitar
        // firmas reutilizables entre redes. No se usa value_usd externo porque
        // aceptar valores no parseados es blind signing inseguro.
        if !self.rules.allowed_chain_ids.contains(&parsed.chain_id) {
            return Err(format!(
                "chain_id {} no autorizado, posible replay cross-chain",
                parsed.chain_id
            ));
        }

        // Reject unsupported Ethereum methods only for Ethereum chain.
        // Bitcoin (BtcRawTx) and Solana (SolanaTransfer) are not rejected by method.
        if parsed.chain == ChainType::Ethereum {
            if let TxMethod::EthUnsupported(sel) = &parsed.method {
                return Err(format!("método de contrato Ethereum no soportado: {}", sel));
            }
        }

        // Reject any generic unsupported method on any chain.
        if let TxMethod::Unsupported(sel) = &parsed.method {
            return Err(format!(
                "método no soportado para la cadena {:?}: {}",
                parsed.chain, sel
            ));
        }

        let effective_volume = self.effective_volume_units(parsed)?;
        let projected_volume = safe_volume_add(self.volume_today_units, effective_volume)?;
        if projected_volume > self.rules.max_daily_volume_units {
            return Err(format!(
                "umbral de volumen excedido desde payload real: limite diario de {} unidades",
                self.rules.max_daily_volume_units
            ));
        }

        for destination in &parsed.destinations {
            match destination.kind {
                ParsedDestinationKind::Erc20Contract => {
                    if !self
                        .rules
                        .allowed_erc20_contracts
                        .contains(&destination.address)
                    {
                        return Err(format!(
                            "contrato ERC-20 no autorizado: {}",
                            destination.address
                        ));
                    }
                }
                ParsedDestinationKind::Erc20Receiver | ParsedDestinationKind::Native => {
                    let is_authorized_dex = self
                        .rules
                        .allowed_dex_contracts
                        .contains(&destination.address);
                    let is_authorized_cex = self
                        .rules
                        .allowed_cex_destinations
                        .contains(&destination.address);

                    if !is_authorized_dex && !is_authorized_cex {
                        return Err(format!(
                            "bloqueo: destino '{}' no autorizado",
                            destination.address
                        ));
                    }
                }
            }
        }

        Ok(())
    }

    pub fn commit_transaction(&mut self, estimated_volume_units: Decimal) -> Result<(), String> {
        let projected = safe_volume_add(self.volume_today_units, estimated_volume_units)?;
        if projected > self.rules.max_daily_volume_units {
            return Err(format!(
                "volumen diario excedido: {} > limite {}",
                projected, self.rules.max_daily_volume_units
            ));
        }
        self.tx_count_today = self.tx_count_today.saturating_add(1);
        self.volume_today_units = projected;
        Ok(())
    }

    pub fn effective_volume_units(&self, parsed: &ParsedTx) -> Result<Decimal, String> {
        let mut total = Decimal::ZERO;
        let erc20_contract = parsed
            .destinations
            .iter()
            .find(|destination| matches!(destination.kind, ParsedDestinationKind::Erc20Contract))
            .map(|destination| destination.address.as_str());

        for destination in &parsed.destinations {
            let volume = match destination.kind {
                ParsedDestinationKind::Erc20Contract => Decimal::ZERO,
                ParsedDestinationKind::Erc20Receiver => {
                    let contract =
                        erc20_contract.ok_or_else(|| "ERC-20 sin contrato parseado".to_string())?;
                    let decimals = match self.rules.token_decimals.get(contract).copied() {
                        Some(decimals) => decimals,
                        None => {
                            return Err(format!(
                                "Token no registrado: se requiere especificar decimales para {}",
                                contract
                            ))
                        }
                    };
                    token_units(destination.value_units, decimals)
                }
                ParsedDestinationKind::Native => {
                    native_units(parsed.chain.clone(), destination.value_units)
                }
            };
            total = safe_volume_add(total, volume)?;
        }
        Ok(total)
    }
}

fn token_units(value_units: u128, decimals: u8) -> Decimal {
    Decimal::from(value_units) / Decimal::from(10u128.pow(u32::from(decimals.min(38))))
}

fn safe_volume_add(left: Decimal, right: Decimal) -> Result<Decimal, String> {
    if left.is_sign_negative() || right.is_sign_negative() {
        return Err("volumen parseado invalido".into());
    }
    let out = left
        .checked_add(right)
        .ok_or_else(|| "suma de volumen no valida: desbordamiento".to_string())?;
    Ok(out)
}

fn now_unix() -> u64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs()
}

/* rol: derivación de claves multiecosistema con zeroize en drop */

use crate::crypto::blockchain::types::{ChainType, SignatureResponse};
use crate::security::vault::derive_vault_key;
use ed25519_dalek::{Signer, SigningKey};
use hmac::{Hmac, Mac};
use secp256k1::{Message, PublicKey, Scalar, Secp256k1, SecretKey};
use sha2::{Digest as Sha2Digest, Sha256, Sha512};
use sha3::Keccak256;
use zeroize::Zeroize;

type HmacSha512 = Hmac<Sha512>;

pub struct UniversalKeymaster {
    master_seed: [u8; 64],
}

impl Drop for UniversalKeymaster {
    fn drop(&mut self) {
        self.master_seed.zeroize();
    }
}

impl UniversalKeymaster {
    pub fn new(cajas: [i64; 5], master_phrase: &str) -> Result<Self, String> {
        let mut vault_key = derive_vault_key(cajas, master_phrase)?;
        let mut hmac = HmacSha512::new_from_slice(b"Livencrypt Universal Seed")
            .map_err(|error| error.to_string())?;
        hmac.update(&vault_key);

        let mut master_seed = [0u8; 64];
        master_seed.copy_from_slice(&hmac.finalize().into_bytes());

        vault_key.zeroize();

        Ok(Self { master_seed })
    }

    fn derive_slip10_child(
        &self,
        chain: &ChainType,
        account: u32,
        index: u32,
    ) -> Result<[u8; 32], String> {
        match chain {
            ChainType::Bitcoin | ChainType::Ethereum => {
                self.derive_bip32_secp256k1_child(coin_id(chain), account, index)
            }
            ChainType::Solana => self.derive_slip10_ed25519_child(account, index),
            ChainType::Monero => Err("Monero requiere derivacion aislada".into()),
        }
    }

    fn derive_bip32_secp256k1_child(
        &self,
        coin_type: u32,
        account: u32,
        index: u32,
    ) -> Result<[u8; 32], String> {
        for candidate_index in index..index.saturating_add(32) {
            let path = [
                hardened(44),
                hardened(coin_type),
                hardened(account),
                0,
                candidate_index,
            ];
            if let Ok(mut key) = derive_bip32_secp_path(&self.master_seed, &path) {
                let out = key;
                key.zeroize();
                return Ok(out);
            }
        }
        Err("No se pudo derivar clave secp256k1 valida en ventana BIP32".into())
    }

    fn derive_slip10_ed25519_child(&self, account: u32, index: u32) -> Result<[u8; 32], String> {
        let path = [
            hardened(44),
            hardened(501),
            hardened(account),
            hardened(0),
            hardened(index),
        ];
        derive_slip10_ed25519_path(&self.master_seed, &path)
    }

    pub fn get_public_key(
        &self,
        chain: &ChainType,
        account: u32,
        index: u32,
    ) -> Result<Vec<u8>, String> {
        let mut raw_key = self.derive_slip10_child(chain, account, index)?;

        let result = (|| -> Result<Vec<u8>, String> {
            match chain {
                ChainType::Bitcoin => {
                    let secp = Secp256k1::new();
                    let sk = SecretKey::from_slice(&raw_key).map_err(|error| error.to_string())?;
                    let pk = PublicKey::from_secret_key(&secp, &sk);
                    Ok(pk.serialize().to_vec())
                }
                ChainType::Ethereum => {
                    let secp = Secp256k1::new();
                    let sk = SecretKey::from_slice(&raw_key).map_err(|error| error.to_string())?;
                    let pk = PublicKey::from_secret_key(&secp, &sk);
                    Ok(pk.serialize_uncompressed()[1..].to_vec())
                }
                ChainType::Solana => {
                    let sk = SigningKey::from_bytes(&raw_key);
                    Ok(sk.verifying_key().to_bytes().to_vec())
                }
                ChainType::Monero => {
                    Err("Monero requiere derivacion multilave aislada no soportada".to_string())
                }
            }
        })();

        raw_key.zeroize();
        result
    }

    pub fn sign_transaction(
        &self,
        chain: &ChainType,
        account: u32,
        index: u32,
        raw_payload: &[u8],
    ) -> Result<SignatureResponse, String> {
        let mut raw_key = self.derive_slip10_child(chain, account, index)?;

        let result = (|| -> Result<SignatureResponse, String> {
            match chain {
                ChainType::Bitcoin | ChainType::Ethereum => {
                    let secp = Secp256k1::new();
                    let sk = SecretKey::from_slice(&raw_key).map_err(|error| error.to_string())?;

                    let tx_hash = match chain {
                        ChainType::Bitcoin => {
                            let hash1 = Sha256::digest(raw_payload);
                            let hash2 = Sha256::digest(hash1);
                            let mut array = [0u8; 32];
                            array.copy_from_slice(&hash2);
                            array
                        }
                        _ => {
                            let hash = Keccak256::digest(raw_payload);
                            let mut array = [0u8; 32];
                            array.copy_from_slice(&hash);
                            array
                        }
                    };

                    let msg = Message::from_digest_slice(&tx_hash)
                        .map_err(|error| error.to_string())?;
                    let recoverable_sig = secp.sign_ecdsa_recoverable(&msg, &sk);
                    let (rec_id, sig_bytes) = recoverable_sig.serialize_compact();
                    let public_key = match chain {
                        ChainType::Bitcoin => PublicKey::from_secret_key(&secp, &sk)
                            .serialize()
                            .to_vec(),
                        _ => PublicKey::from_secret_key(&secp, &sk)
                            .serialize_uncompressed()[1..]
                            .to_vec(),
                    };

                    Ok(SignatureResponse {
                        public_key,
                        raw_signature: sig_bytes.to_vec(),
                        recovery_id: Some(rec_id.to_i32() as u8),
                    })
                }
                ChainType::Solana => {
                    let sk = SigningKey::from_bytes(&raw_key);
                    let sig = sk.sign(raw_payload);

                    Ok(SignatureResponse {
                        public_key: sk.verifying_key().to_bytes().to_vec(),
                        raw_signature: sig.to_bytes().to_vec(),
                        recovery_id: None,
                    })
                }
                ChainType::Monero => Err(
                    "Monero requiere un esquema especializado de firmas en anillo y payload multilave especifico".to_string(),
                ),
            }
        })();

        raw_key.zeroize();
        result
    }
}

fn hardened(index: u32) -> u32 {
    index | 0x8000_0000
}

fn hmac_sha512_bytes(key: &[u8], data: &[u8]) -> Result<[u8; 64], String> {
    let mut hmac = HmacSha512::new_from_slice(key).map_err(|error| error.to_string())?;
    hmac.update(data);
    let bytes = hmac.finalize().into_bytes();
    let mut out = [0u8; 64];
    out.copy_from_slice(&bytes);
    Ok(out)
}

fn derive_bip32_secp_path(seed: &[u8; 64], path: &[u32]) -> Result<[u8; 32], String> {
    let mut master = hmac_sha512_bytes(b"Bitcoin seed", seed)?;
    let mut key = [0u8; 32];
    key.copy_from_slice(&master[..32]);
    let mut chain_code = [0u8; 32];
    chain_code.copy_from_slice(&master[32..]);
    master.zeroize();

    SecretKey::from_slice(&key)
        .map_err(|error| format!("master BIP32 secp256k1 invalida: {}", error))?;

    let secp = Secp256k1::new();
    for child_index in path {
        let mut data = Vec::with_capacity(37);
        if child_index & 0x8000_0000 != 0 {
            data.push(0);
            data.extend_from_slice(&key);
        } else {
            let parent = SecretKey::from_slice(&key)
                .map_err(|error| format!("clave padre BIP32 invalida: {}", error))?;
            let public = PublicKey::from_secret_key(&secp, &parent).serialize();
            data.extend_from_slice(&public);
        }
        data.extend_from_slice(&child_index.to_be_bytes());

        let mut material = hmac_sha512_bytes(&chain_code, &data)?;
        data.zeroize();

        let mut tweak_bytes = [0u8; 32];
        tweak_bytes.copy_from_slice(&material[..32]);
        let tweak = Scalar::from_be_bytes(tweak_bytes).map_err(|_| "tweak BIP32 fuera de rango")?;
        let parent = SecretKey::from_slice(&key)
            .map_err(|error| format!("clave padre BIP32 invalida: {}", error))?;
        let child = parent
            .add_tweak(&tweak)
            .map_err(|error| format!("clave hija BIP32 invalida: {}", error))?;
        key.copy_from_slice(child.as_ref());
        chain_code.copy_from_slice(&material[32..]);
        tweak_bytes.zeroize();
        material.zeroize();
    }

    chain_code.zeroize();
    Ok(key)
}

fn derive_slip10_ed25519_path(seed: &[u8; 64], path: &[u32]) -> Result<[u8; 32], String> {
    let mut master = hmac_sha512_bytes(b"ed25519 seed", seed)?;
    let mut key = [0u8; 32];
    key.copy_from_slice(&master[..32]);
    let mut chain_code = [0u8; 32];
    chain_code.copy_from_slice(&master[32..]);
    master.zeroize();

    for child_index in path {
        if child_index & 0x8000_0000 == 0 {
            return Err("SLIP10 ed25519 solo admite indices endurecidos".into());
        }
        let mut data = Vec::with_capacity(37);
        data.push(0);
        data.extend_from_slice(&key);
        data.extend_from_slice(&child_index.to_be_bytes());
        let mut material = hmac_sha512_bytes(&chain_code, &data)?;
        data.zeroize();
        key.copy_from_slice(&material[..32]);
        chain_code.copy_from_slice(&material[32..]);
        material.zeroize();
    }

    chain_code.zeroize();
    Ok(key)
}

fn coin_id(chain: &ChainType) -> u32 {
    match chain {
        ChainType::Bitcoin => 0,
        ChainType::Ethereum => 60,
        ChainType::Solana => 501,
        // Monero no alcanza aquí porque derive_slip10_child retorna Err antes.
        // Valor documental para futuras cadenas BIP-44.
        ChainType::Monero => 128,
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later

/* rol: host de mensajeria nativa seguro para extension web stateless */

use crate::crypto::blockchain::types::{KeyDerivationRequest, TransactionSignRequest};
use serde::{Deserialize, Serialize};
use std::io::{self, Read, Write};
use zeroize::Zeroize;

const MAX_NATIVE_MESSAGE_BYTES: usize = 1024 * 1024;

#[derive(Serialize, Deserialize)]
#[serde(tag = "type", content = "payload")]
pub enum ExtensionRequest {
    DeriveKey(KeyDerivationRequest),
    SignTransaction(TransactionSignRequest),
}

#[derive(Serialize, Deserialize)]
#[serde(tag = "status", content = "data")]
pub enum ExtensionResponse<T> {
    Success(T),
    Error(String),
}

pub struct NativeProxyChannel;

impl NativeProxyChannel {
    pub fn read_next_request() -> Result<ExtensionRequest, String> {
        let mut length_buffer = [0u8; 4];
        io::stdin()
            .read_exact(&mut length_buffer)
            .map_err(|error| format!("error al leer longitud del mensaje: {}", error))?;

        let message_length = u32::from_le_bytes(length_buffer) as usize;
        if message_length > MAX_NATIVE_MESSAGE_BYTES {
            return Err("longitud de mensaje excede el limite de seguridad de 1MB".to_string());
        }

        let mut json_buffer = vec![0u8; message_length];
        io::stdin()
            .read_exact(&mut json_buffer)
            .map_err(|error| format!("error al leer cuerpo del mensaje: {}", error))?;

        let request = serde_json::from_slice(&json_buffer)
            .map_err(|error| format!("error de deserializacion estructurada: {}", error))?;

        json_buffer.zeroize();
        Ok(request)
    }

    pub fn write_response<T: Serialize>(response: &ExtensionResponse<T>) -> Result<(), String> {
        let json_bytes = serde_json::to_vec(response)
            .map_err(|error| format!("error de serializacion de respuesta: {}", error))?;
        write_stdout_native_message(&json_bytes)
    }
}

pub fn read_native_message<R: Read>(reader: &mut R) -> Result<Vec<u8>, String> {
    let mut length_buffer = [0u8; 4];
    reader
        .read_exact(&mut length_buffer)
        .map_err(|error| format!("error al leer longitud del mensaje: {}", error))?;

    let message_length = u32::from_le_bytes(length_buffer) as usize;
    if message_length > MAX_NATIVE_MESSAGE_BYTES {
        return Err("longitud de mensaje excede el limite de seguridad de 1MB".to_string());
    }

    let mut payload = vec![0u8; message_length];
    reader
        .read_exact(&mut payload)
        .map_err(|error| format!("error al leer cuerpo del mensaje: {}", error))?;
    Ok(payload)
}

pub fn write_native_message<W: Write>(writer: &mut W, payload: &[u8]) -> Result<(), String> {
    if payload.len() > MAX_NATIVE_MESSAGE_BYTES {
        return Err("respuesta excede el limite de seguridad de 1MB".to_string());
    }

    let length_prefix = (payload.len() as u32).to_le_bytes();
    writer
        .write_all(&length_prefix)
        .map_err(|error| format!("error al escribir prefijo de longitud: {}", error))?;
    writer
        .write_all(payload)
        .map_err(|error| format!("error al escribir cuerpo de la respuesta: {}", error))?;
    writer
        .flush()
        .map_err(|error| format!("error al vaciar el canal de salida: {}", error))
}

fn write_stdout_native_message(payload: &[u8]) -> Result<(), String> {
    let mut stdout = io::stdout();
    write_native_message(&mut stdout, payload)
}

// SPDX-License-Identifier: GPL-3.0-or-later
// Copyright (C) 2026 Synergia86 <livencrypt1@gmail.com>
/* rol: supervisor maestro y coordinador de la pasarela de custodia universal */

use crate::crypto::blockchain::breaker::{GatewaySecurityState, TacticalCircuitBreaker};
use crate::crypto::blockchain::firewall::LiveFirewall;
use crate::crypto::blockchain::multi_curve::UniversalKeymaster;
use crate::crypto::blockchain::proxy::{
    read_native_message, write_native_message, ExtensionRequest, ExtensionResponse,
};
use crate::crypto::blockchain::tx_parser::parse_tx_payload;
use std::path::Path;
use std::time::{SystemTime, UNIX_EPOCH};

#[cfg(unix)]
use std::os::unix::net::UnixListener;

pub struct GatewaySupervisor {
    pub keymaster: UniversalKeymaster,
    pub firewall: LiveFirewall,
    pub breaker: TacticalCircuitBreaker,
}

impl GatewaySupervisor {
    pub fn new(keymaster: UniversalKeymaster, firewall: LiveFirewall) -> Self {
        Self {
            keymaster,
            firewall,
            breaker: TacticalCircuitBreaker::new(),
        }
    }

    pub fn run_loop_iteration(&mut self, request: ExtensionRequest) -> Result<Vec<u8>, String> {
        if self.breaker.state == GatewaySecurityState::HardLocked {
            let response = ExtensionResponse::<()>::Error(
                "acceso revocado: intervencion local requerida".to_string(),
            );
            return serde_json::to_vec(&response).map_err(|error| error.to_string());
        }

        let now = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();

        match request {
            ExtensionRequest::DeriveKey(derivation) => {
                if matches!(self.breaker.state, GatewaySecurityState::Frozen { .. }) {
                    let response = ExtensionResponse::<Vec<u8>>::Error(
                        "operacion suspendida temporalmente".to_string(),
                    );
                    return serde_json::to_vec(&response).map_err(|error| error.to_string());
                }

                let response = match self.keymaster.get_public_key(
                    &derivation.chain,
                    derivation.account_index,
                    derivation.address_index,
                ) {
                    Ok(public_key) => ExtensionResponse::Success(public_key),
                    Err(error) => ExtensionResponse::<Vec<u8>>::Error(error),
                };
                serde_json::to_vec(&response).map_err(|error| error.to_string())
            }
            ExtensionRequest::SignTransaction(sign_request) => {
                let parsed =
                    match parse_tx_payload(&sign_request.chain, &sign_request.raw_tx_payload) {
                        Ok(parsed) => parsed,
                        Err(error) => {
                            let error = format!("no se pudo parsear payload: {}", error);
                            self.register_integrity_failure(&error, now);
                            let response = ExtensionResponse::<()>::Error(error);
                            return serde_json::to_vec(&response)
                                .map_err(|error| error.to_string());
                        }
                    };

                if let Err(firewall_error) = self.firewall.validate_parsed_transaction(&parsed) {
                    self.register_integrity_failure(&firewall_error, now);
                    let response = ExtensionResponse::<()>::Error(format!(
                        "violacion de perimetro: {}",
                        firewall_error
                    ));
                    return serde_json::to_vec(&response).map_err(|error| error.to_string());
                }
                let effective_volume = match self.firewall.effective_volume_units(&parsed) {
                    Ok(volume) => volume,
                    Err(error) => {
                        self.register_integrity_failure(&error, now);
                        let response = ExtensionResponse::<()>::Error(error);
                        return serde_json::to_vec(&response).map_err(|error| error.to_string());
                    }
                };

                // Firmar primero para no desperdiciar contadores si la firma falla.
                let signature_response = match self.keymaster.sign_transaction(
                    &sign_request.chain,
                    sign_request.account_index,
                    sign_request.address_index,
                    &sign_request.raw_tx_payload,
                ) {
                    Ok(signature) => signature,
                    Err(error) => {
                        let response = ExtensionResponse::<()>::Error(error);
                        return serde_json::to_vec(&response).map_err(|error| error.to_string());
                    }
                };

                // Commit después de firma exitosa para que fallas de firma no desperdicien contadores.
                if let Err(commit_error) = self.firewall.commit_transaction(effective_volume) {
                    self.register_integrity_failure(&commit_error, now);
                    let response = ExtensionResponse::<()>::Error(format!(
                        "commit rechazado: {}",
                        commit_error
                    ));
                    return serde_json::to_vec(&response).map_err(|error| error.to_string());
                }

                let response = ExtensionResponse::Success(signature_response);
                serde_json::to_vec(&response).map_err(|error| error.to_string())
            }
        }
    }

    fn register_integrity_failure(&mut self, error: &str, now: u64) {
        self.breaker.integrity_failures = self.breaker.integrity_failures.saturating_add(1);
        self.breaker.trip_breaker(error, now);
        if self.breaker.integrity_failures >= 3 {
            self.breaker.trigger_hard_lock();
        }
    }

    pub fn handle_native_payload(&mut self, payload: &[u8]) -> Vec<u8> {
        let request = match serde_json::from_slice::<ExtensionRequest>(payload) {
            Ok(request) => request,
            Err(error) => {
                return error_response_bytes(format!(
                    "error de deserializacion estructurada: {}",
                    error
                ));
            }
        };

        self.run_loop_iteration(request)
            .unwrap_or_else(error_response_bytes)
    }

    #[cfg(unix)]
    pub fn serve_unix_socket(&mut self, socket_path: &Path) -> Result<(), String> {
        if let Some(parent) = socket_path.parent() {
            std::fs::create_dir_all(parent)
                .map_err(|error| format!("no se pudo crear directorio IPC: {}", error))?;
        }

        if socket_path.exists() {
            std::fs::remove_file(socket_path)
                .map_err(|error| format!("no se pudo limpiar socket IPC previo: {}", error))?;
        }

        let listener = UnixListener::bind(socket_path)
            .map_err(|error| format!("no se pudo abrir socket IPC: {}", error))?;

        for stream in listener.incoming() {
            let mut stream =
                stream.map_err(|error| format!("conexion IPC entrante invalida: {}", error))?;
            let _ = stream.set_read_timeout(Some(std::time::Duration::from_millis(200)));
            let _ = stream.set_write_timeout(Some(std::time::Duration::from_millis(200)));
            let payload = read_native_message(&mut stream)?;
            let response = self.handle_native_payload(&payload);
            write_native_message(&mut stream, &response)?;
        }

        Ok(())
    }
}

fn error_response_bytes(error: String) -> Vec<u8> {
    serde_json::to_vec(&ExtensionResponse::<()>::Error(error)).unwrap_or_else(|_| {
        b"{\"status\":\"Error\",\"data\":\"error interno de serializacion\"}".to_vec()
    })
}

// SPDX-License-Identifier: GPL-3.0-or-later
// Copyright (C) 2026 Synergia86 <livencrypt1@gmail.com>
/* rol: parser forense minimo de transacciones antes de firma */

use crate::crypto::blockchain::types::{ChainType, TxMethod};
use rust_decimal::Decimal;
use serde::Deserialize;

#[derive(Clone, Debug, PartialEq)]
pub enum ParsedDestinationKind {
    Native,
    Erc20Contract,
    Erc20Receiver,
}

#[derive(Clone, Debug, PartialEq)]
pub struct ParsedDestination {
    pub address: String,
    pub value_units: u128,
    pub kind: ParsedDestinationKind,
}

#[derive(Clone, Debug, PartialEq)]
pub struct ParsedTx {
    pub chain: ChainType,
    pub to_address: String,
    pub value_units: u128,
    /// Volumen legacy calculado en la cadena de origen. No utilizado por el cortafuegos;
    /// usar `Firewall::effective_volume_units()` para cálculo de volumen diario.
    pub volume_units: Decimal,
    pub destinations: Vec<ParsedDestination>,
    pub chain_id: u64,
    pub method: TxMethod,
    pub summary: String,
}

pub fn parse_tx_payload(chain: &ChainType, raw_tx_payload: &[u8]) -> Result<ParsedTx, String> {
    match chain {
        ChainType::Ethereum => parse_ethereum_payload(raw_tx_payload),
        ChainType::Bitcoin => parse_bitcoin_payload(raw_tx_payload),
        ChainType::Solana => parse_solana_gateway_payload(raw_tx_payload),
        ChainType::Monero => Err("Monero requiere parser forense especifico no integrado".into()),
    }
}

fn parse_ethereum_payload(raw: &[u8]) -> Result<ParsedTx, String> {
    if raw.is_empty() {
        return Err("payload Ethereum vacio".into());
    }

    let (tx_type, fields) = if raw[0] == 0x02 {
        (Some(0x02), parse_rlp_list(&raw[1..])?)
    } else if raw[0] == 0x01 {
        (Some(0x01), parse_rlp_list(&raw[1..])?)
    } else {
        (None, parse_rlp_list(raw)?)
    };

    let (chain_id, to, value, data) = match tx_type {
        Some(0x02) => {
            if fields.len() < 9 {
                return Err("EIP-1559 incompleto".into());
            }
            (
                bytes_to_u64(fields[0])?,
                fields[5],
                bytes_to_u128(fields[6])?,
                fields[7],
            )
        }
        Some(0x01) => {
            if fields.len() < 8 {
                return Err("EIP-2930 incompleto".into());
            }
            (
                bytes_to_u64(fields[0])?,
                fields[4],
                bytes_to_u128(fields[5])?,
                fields[6],
            )
        }
        _ => {
            if fields.len() < 9 {
                return Err("transaccion Ethereum legacy incompleta".into());
            }
            let v = bytes_to_u64(fields[6])?;
            let chain_id = if v >= 35 {
                (v - 35) / 2
            } else {
                return Err("Ethereum legacy sin chain_id EIP-155".into());
            };
            (chain_id, fields[3], bytes_to_u128(fields[4])?, fields[5])
        }
    };

    if to.len() != 20 {
        return Err("Ethereum sin destino externo de 20 bytes".into());
    }
    let contract_or_to = format!("0x{}", hex::encode(to));
    let (to_address, value_units, method) =
        if data.starts_with(&[0xa9, 0x05, 0x9c, 0xbb]) && data.len() >= 68 {
            let token_receiver = format!("0x{}", hex::encode(&data[16..36]));
            let amount = bytes_to_u128_word(&data[36..68])?;
            (token_receiver.clone(), amount, TxMethod::EthErc20Transfer)
        } else if data.is_empty() {
            (contract_or_to.clone(), value, TxMethod::EthTransfer)
        } else {
            (
                contract_or_to.clone(),
                value,
                TxMethod::EthUnsupported(format!("0x{}", hex::encode(&data[..4]))),
            )
        };

    let destinations = if let TxMethod::EthErc20Transfer = method {
        let token_receiver = format!("0x{}", hex::encode(&data[16..36]));
        let amount = bytes_to_u128_word(&data[36..68])?;
        vec![
            ParsedDestination {
                address: contract_or_to.clone(),
                value_units: 0,
                kind: ParsedDestinationKind::Erc20Contract,
            },
            ParsedDestination {
                address: token_receiver,
                value_units: amount,
                kind: ParsedDestinationKind::Erc20Receiver,
            },
        ]
    } else {
        vec![ParsedDestination {
            address: contract_or_to.clone(),
            value_units: value,
            kind: ParsedDestinationKind::Native,
        }]
    };

    let summary = match &method {
        TxMethod::EthTransfer => {
            format!("transferencia_eth to={} value={}", to_address, value_units)
        }
        TxMethod::EthErc20Transfer => format!(
            "erc20_transfer contrato={} value={}",
            contract_or_to, value_units
        ),
        TxMethod::EthUnsupported(sel) => format!(
            "unsupported_method {} to={} value={}",
            sel, contract_or_to, value_units
        ),
        TxMethod::BtcRawTx => format!("bitcoin_raw_tx to={} value={}", to_address, value_units),
        TxMethod::SolanaTransfer => {
            format!("solana_transfer to={} value={}", to_address, value_units)
        }
        TxMethod::Unsupported(sel) => format!(
            "unsupported_chain_method {} to={} value={}",
            sel, to_address, value_units
        ),
    };

    Ok(ParsedTx {
        chain: ChainType::Ethereum,
        to_address,
        value_units,
        volume_units: Decimal::ZERO,
        destinations,
        chain_id,
        method,
        summary,
    })
}

fn parse_bitcoin_payload(raw: &[u8]) -> Result<ParsedTx, String> {
    let mut reader = ByteReader::new(raw);
    reader.take(4)?;
    let input_count = reader.read_varint()?;
    if input_count == 0 || input_count > 10_000 {
        return Err("Bitcoin inputs invalidos".into());
    }
    for _ in 0..input_count {
        reader.take(32)?;
        reader.take(4)?;
        let script_len = reader.read_varint()?;
        if script_len > 10_000 {
            return Err("script Bitcoin demasiado largo".into());
        }
        reader.take_usize(script_len)?;
        reader.take(4)?;
    }

    let output_count = reader.read_varint()?;
    if output_count == 0 || output_count > 50 {
        return Err("Bitcoin outputs invalidos (limite 50)".into());
    }

    let mut destinations = Vec::new();
    let mut total_value = 0u128;
    for output_index in 0..output_count {
        let value = reader.read_u64_le()? as u128;
        let script_len = reader.read_varint()?;
        if script_len > 10_000 {
            return Err("script Bitcoin demasiado largo".into());
        }
        let script = reader.take_usize(script_len)?;
        let address = bitcoin_script_destination(script).ok_or_else(|| {
            format!(
                "Bitcoin script tipo no soportado en output {}",
                output_index
            )
        })?;
        total_value = total_value.saturating_add(value);
        destinations.push(ParsedDestination {
            address,
            value_units: value,
            kind: ParsedDestinationKind::Native,
        });
    }
    reader.take(4)?;

    let to_address = destinations
        .first()
        .map(|destination| destination.address.clone())
        .ok_or_else(|| "Bitcoin sin outputs".to_string())?;
    let volume_units = native_units(ChainType::Bitcoin, total_value);
    let summary = format!(
        "Bitcoin outputs={} total_sats={} volume_units={:.8}",
        output_count, total_value, volume_units
    );
    Ok(ParsedTx {
        chain: ChainType::Bitcoin,
        to_address,
        value_units: total_value,
        volume_units,
        destinations,
        chain_id: 0,
        method: TxMethod::BtcRawTx,
        summary,
    })
}

fn parse_solana_gateway_payload(raw: &[u8]) -> Result<ParsedTx, String> {
    #[derive(Deserialize)]
    struct SolanaPayload {
        to_address: String,
        lamports: u64,
        method: Option<String>,
    }

    let payload: SolanaPayload = serde_json::from_slice(raw)
        .map_err(|error| format!("payload Solana no es JSON de pasarela: {}", error))?;
    if payload.to_address.trim().is_empty() {
        return Err("Solana sin destino".into());
    }
    if payload.lamports > 100_000_000_000_000u64 {
        return Err("Solana lamports excesivos (limite 10^14)".into());
    }
    let method_str = payload.method.as_deref().unwrap_or("transfer");
    let method = if method_str == "transfer" {
        TxMethod::SolanaTransfer
    } else {
        TxMethod::Unsupported(method_str.to_string())
    };
    let summary = format!(
        "Solana destino={} lamports={} metodo={}",
        payload.to_address, payload.lamports, method_str
    );
    Ok(ParsedTx {
        chain: ChainType::Solana,
        to_address: payload.to_address.clone(),
        value_units: payload.lamports as u128,
        volume_units: native_units(ChainType::Solana, payload.lamports as u128),
        destinations: vec![ParsedDestination {
            address: payload.to_address,
            value_units: payload.lamports as u128,
            kind: ParsedDestinationKind::Native,
        }],
        chain_id: 501,
        method,
        summary,
    })
}

fn parse_rlp_list(input: &[u8]) -> Result<Vec<&[u8]>, String> {
    parse_rlp_list_inner(input, 0)
}

fn parse_rlp_list_inner(input: &[u8], depth: usize) -> Result<Vec<&[u8]>, String> {
    const MAX_RLP_DEPTH: usize = 16;
    if depth > MAX_RLP_DEPTH {
        return Err("RLP excede profundidad maxima de recursion".into());
    }
    let (payload, consumed) = parse_rlp_payload(input, true)?;
    if consumed != input.len() {
        return Err("RLP con bytes sobrantes".into());
    }
    let mut fields = Vec::new();
    let mut offset = 0usize;
    while offset < payload.len() {
        let (field, consumed) = parse_rlp_payload(&payload[offset..], false)?;
        fields.push(field);
        offset += consumed;
    }
    Ok(fields)
}

fn parse_rlp_payload(input: &[u8], expect_list: bool) -> Result<(&[u8], usize), String> {
    let Some(&prefix) = input.first() else {
        return Err("RLP vacio".into());
    };
    match prefix {
        0x00..=0x7f if !expect_list => Ok((&input[..1], 1)),
        0x80..=0xb7 if !expect_list => {
            let len = (prefix - 0x80) as usize;
            bounded_slice(input, 1, len).map(|value| (value, 1 + len))
        }
        0xb8..=0xbf if !expect_list => {
            let len_len = (prefix - 0xb7) as usize;
            let len = read_be_len(bounded_slice(input, 1, len_len)?)?;
            bounded_slice(input, 1 + len_len, len).map(|value| (value, 1 + len_len + len))
        }
        0xc0..=0xf7 if expect_list => {
            let len = (prefix - 0xc0) as usize;
            bounded_slice(input, 1, len).map(|value| (value, 1 + len))
        }
        0xf8..=0xff if expect_list => {
            let len_len = (prefix - 0xf7) as usize;
            let len = read_be_len(bounded_slice(input, 1, len_len)?)?;
            bounded_slice(input, 1 + len_len, len).map(|value| (value, 1 + len_len + len))
        }
        _ => Err("RLP tipo inesperado".into()),
    }
}

fn bounded_slice(input: &[u8], start: usize, len: usize) -> Result<&[u8], String> {
    input
        .get(start..start.saturating_add(len))
        .ok_or_else(|| "RLP truncado".to_string())
}

fn read_be_len(bytes: &[u8]) -> Result<usize, String> {
    if bytes.is_empty() || bytes.len() > std::mem::size_of::<usize>() {
        return Err("longitud RLP invalida".into());
    }
    let mut out = 0usize;
    for byte in bytes {
        out = (out << 8) | (*byte as usize);
    }
    Ok(out)
}

fn bytes_to_u64(bytes: &[u8]) -> Result<u64, String> {
    if bytes.len() > 8 {
        return Err("entero demasiado grande para u64".into());
    }
    let mut out = 0u64;
    for byte in bytes {
        out = (out << 8) | (*byte as u64);
    }
    Ok(out)
}

fn bytes_to_u128(bytes: &[u8]) -> Result<u128, String> {
    if bytes.len() > 16 {
        return Err("entero demasiado grande para u128".into());
    }
    let mut out = 0u128;
    for byte in bytes {
        out = (out << 8) | (*byte as u128);
    }
    Ok(out)
}

fn bytes_to_u128_word(bytes: &[u8]) -> Result<u128, String> {
    if bytes.len() > 32 {
        return Err("uint256 demasiado largo".into());
    }
    if bytes.len() > 16 && bytes[..bytes.len() - 16].iter().any(|byte| *byte != 0) {
        return Err(
            "valor de token excede 128 bits; no soportado (ERC-20 con mas de 128 bits)".into(),
        );
    }
    let start = bytes.len().saturating_sub(16);
    bytes_to_u128(&bytes[start..])
}

fn bitcoin_script_destination(script: &[u8]) -> Option<String> {
    if script.len() == 25
        && script[0] == 0x76
        && script[1] == 0xa9
        && script[2] == 0x14
        && script[23] == 0x88
        && script[24] == 0xac
    {
        return Some(format!("btc:p2pkh:{}", hex::encode(&script[3..23])));
    }
    if script.len() == 23 && script[0] == 0xa9 && script[1] == 0x14 && script[22] == 0x87 {
        return Some(format!("btc:p2sh:{}", hex::encode(&script[2..22])));
    }
    if script.len() == 22 && script[0] == 0x00 && script[1] == 0x14 {
        return Some(format!("btc:p2wpkh:{}", hex::encode(&script[2..22])));
    }
    if script.len() == 34 && script[0] == 0x00 && script[1] == 0x20 {
        return Some(format!("btc:p2wsh:{}", hex::encode(&script[2..34])));
    }
    None
}

pub fn native_units(chain: ChainType, value_units: u128) -> Decimal {
    match chain {
        ChainType::Ethereum => {
            Decimal::from(value_units) / Decimal::from(1_000_000_000_000_000_000u128)
        }
        ChainType::Bitcoin => Decimal::from(value_units) / Decimal::from(100_000_000u64),
        ChainType::Solana => Decimal::from(value_units) / Decimal::from(1_000_000_000u64),
        ChainType::Monero => Decimal::from(value_units),
    }
}

struct ByteReader<'a> {
    bytes: &'a [u8],
    offset: usize,
}

impl<'a> ByteReader<'a> {
    fn new(bytes: &'a [u8]) -> Self {
        Self { bytes, offset: 0 }
    }

    fn take(&mut self, len: usize) -> Result<&'a [u8], String> {
        let end = self
            .offset
            .checked_add(len)
            .ok_or_else(|| "offset fuera de rango".to_string())?;
        let slice = self
            .bytes
            .get(self.offset..end)
            .ok_or_else(|| "payload truncado".to_string())?;
        self.offset = end;
        Ok(slice)
    }

    fn take_usize(&mut self, len: u64) -> Result<&'a [u8], String> {
        let len = usize::try_from(len).map_err(|_| "longitud excesiva".to_string())?;
        self.take(len)
    }

    fn read_u64_le(&mut self) -> Result<u64, String> {
        let bytes = self.take(8)?;
        let mut array = [0u8; 8];
        array.copy_from_slice(bytes);
        Ok(u64::from_le_bytes(array))
    }

    fn read_varint(&mut self) -> Result<u64, String> {
        let tag = self.take(1)?[0];
        match tag {
            0x00..=0xfc => Ok(tag as u64),
            0xfd => {
                let mut array = [0u8; 2];
                array.copy_from_slice(self.take(2)?);
                Ok(u16::from_le_bytes(array) as u64)
            }
            0xfe => {
                let mut array = [0u8; 4];
                array.copy_from_slice(self.take(4)?);
                Ok(u32::from_le_bytes(array) as u64)
            }
            0xff => self.read_u64_le(),
        }
    }
}

pub enum Web3WorkerCmd {
    UpdateRules {
        max_tx: Option<u32>,
        max_vol: Option<Decimal>,
    },

pub enum Web3WorkerEvent {
    StatusChanged(GatewaySecurityState),
    MetricsUpdated {
        tx_count: u32,
        volume_units: Decimal,
    },
    SyncRules(FirewallRules),
    Notification(String),
}

pub struct Web3BackgroundWorker;

impl Web3BackgroundWorker {
    pub fn spawn(
        mut supervisor: GatewaySupervisor,
    ) -> (Sender<Web3WorkerCmd>, Receiver<Web3WorkerEvent>) {
        let (cmd_tx, cmd_rx) = channel::<Web3WorkerCmd>();
        let (event_tx, event_rx) = channel::<Web3WorkerEvent>();
        let running = Arc::new(AtomicBool::new(true));

        spawn_socket_reader(cmd_tx.clone(), event_tx.clone(), Arc::clone(&running));

        thread::spawn(move || {
            let mut last_tx_count = 0;
            let mut last_volume_units = Decimal::ZERO;

            loop {
                match cmd_rx.recv_timeout(Duration::from_millis(200)) {
                    Ok(Web3WorkerCmd::Shutdown) => {
                        running.store(false, Ordering::Relaxed);
                        break;
                    }
                    Ok(cmd) => process_command(cmd, &mut supervisor, &event_tx),
                    Err(RecvTimeoutError::Timeout) => {}
                    Err(RecvTimeoutError::Disconnected) => break,
                }

                let current_tx_count = supervisor.firewall.tx_count_today;
                let current_volume_units = supervisor.firewall.volume_today_units;
                let volume_changed = (current_volume_units - last_volume_units).abs()
                    > Decimal::from_str_radix("0.00000001", 10).unwrap_or(Decimal::ZERO);
                if current_tx_count != last_tx_count || volume_changed {
                    let _ = event_tx.send(Web3WorkerEvent::MetricsUpdated {
                        tx_count: current_tx_count,
                        volume_units: current_volume_units,
                    });
                    last_tx_count = current_tx_count;
                    last_volume_units = current_volume_units;
                }
            }
        });

        (cmd_tx, event_rx)
    }
}

fn spawn_socket_reader(
    cmd_tx: Sender<Web3WorkerCmd>,
    event_tx: Sender<Web3WorkerEvent>,
    running: Arc<AtomicBool>,
) {
    thread::spawn(move || {
        let socket_path = match resolve_socket_path() {
            Ok(path) => path,
            Err(error) => {
                let _ = event_tx.send(Web3WorkerEvent::Notification(error));
                return;
            }
        };
        if let Err(error) = remove_existing_socket_safely(&socket_path) {
            let _ = event_tx.send(Web3WorkerEvent::Notification(error));
            return;
        }
        let listener = match UnixListener::bind(&socket_path) {
            Ok(listener) => listener,
            Err(error) => {
                let _ = event_tx.send(Web3WorkerEvent::Notification(format!(
                    "No se pudo abrir socket Web3: {}",
                    error
                )));
                return;
            }
        };
        if let Err(error) =
            std::fs::set_permissions(&socket_path, std::fs::Permissions::from_mode(0o600))
        {
            let _ = event_tx.send(Web3WorkerEvent::Notification(format!(
                "Socket Web3 creado, pero no se pudieron ajustar permisos: {}",
                error
            )));
        }

        let _ = event_tx.send(Web3WorkerEvent::Notification(format!(
            "Pasarela Web3 escuchando en {}",
            socket_path.display()
        )));

        let active_connections = Arc::new(AtomicUsize::new(0));
        while running.load(Ordering::Relaxed) {
            let Ok((mut stream, _addr)) = listener.accept() else {
                thread::sleep(Duration::from_millis(50));
                continue;
            };

            #[cfg(unix)]
            if ensure_peer_is_current_user(&stream).is_err() {
                continue;
            }

            if active_connections.load(Ordering::Relaxed) >= MAX_CONCURRENT_CONNECTIONS {
                continue;
            }

            active_connections.fetch_add(1, Ordering::Relaxed);
            let thread_tx = cmd_tx.clone();
            let counter = Arc::clone(&active_connections);

            thread::spawn(move || {
                let _ =
                    stream.set_read_timeout(Some(Duration::from_millis(SOCKET_READ_TIMEOUT_MS)));
                let _ =
                    stream.set_write_timeout(Some(Duration::from_millis(SOCKET_READ_TIMEOUT_MS)));
                if let Ok(payload) = read_native_message(&mut stream) {
                    let _ = thread_tx.send(Web3WorkerCmd::IncomingWebRequest(payload, stream));
                }
                counter.fetch_sub(1, Ordering::Relaxed);
            });
        }

        let _ = std::fs::remove_file(&socket_path);
    });
}

fn process_command(
    cmd: Web3WorkerCmd,
    supervisor: &mut GatewaySupervisor,
    event_tx: &Sender<Web3WorkerEvent>,
) {
    match cmd {
        Web3WorkerCmd::UpdateRules { max_tx, max_vol } => {
            let mut updated = false;
            let mut errors = Vec::new();
            if let Some(tx) = max_tx {
                if tx > 0 {
                    supervisor.firewall.rules.max_daily_tx = tx;
                    updated = true;
                } else {
                    errors.push("max_daily_tx debe ser mayor a 0");
                }
            }
            if let Some(volume) = max_vol {
                if volume > Decimal::ZERO {
                    supervisor.firewall.rules.max_daily_volume_units = volume;
                    updated = true;
                } else {
                    errors.push("max_daily_volume_units debe ser mayor a 0");
                }
            }

            let message = if errors.is_empty() && updated {
                "Limites del cortafuegos actualizados".to_string()
            } else if errors.is_empty() {
                "Error: formato numerico invalido".to_string()
            } else {
                errors.join("; ")
            };
            let _ = event_tx.send(Web3WorkerEvent::Notification(message));
            let _ = event_tx.send(Web3WorkerEvent::SyncRules(
                supervisor.firewall.rules.clone(),
            ));
        }
        Web3WorkerCmd::AddDexContract(mut contract) => {
            normalize_id(&mut contract);
            if contract.is_empty() {
                return;
            }
            if supervisor.firewall.rules.allowed_dex_contracts.len() >= MAX_WHITELIST_ITEMS {
                let _ = event_tx.send(Web3WorkerEvent::Notification(
                    "Lista DEX completa: limite maximo alcanzado".to_string(),
                ));
                return;
            }
            if !supervisor
                .firewall
                .rules
                .allowed_dex_contracts
                .contains(&contract)
            {
                supervisor
                    .firewall
                    .rules
                    .allowed_dex_contracts
                    .push(contract);
                let _ = event_tx.send(Web3WorkerEvent::SyncRules(
                    supervisor.firewall.rules.clone(),
                ));
            }
        }
        Web3WorkerCmd::AddCexDestination(mut destination) => {
            normalize_id(&mut destination);
            if destination.is_empty() {
                return;
            }
            if supervisor.firewall.rules.allowed_cex_destinations.len() >= MAX_WHITELIST_ITEMS {
                let _ = event_tx.send(Web3WorkerEvent::Notification(
                    "Lista CEX completa: limite maximo alcanzado".to_string(),
                ));
                return;
            }
            if !supervisor
                .firewall
                .rules
                .allowed_cex_destinations
                .contains(&destination)
            {
                supervisor
                    .firewall
                    .rules
                    .allowed_cex_destinations
                    .push(destination);
                let _ = event_tx.send(Web3WorkerEvent::SyncRules(
                    supervisor.firewall.rules.clone(),
                ));
            }
        }
        Web3WorkerCmd::ResetBreaker(cajas, mut phrase_bytes) => {
            const MAX_PHRASE_BYTES: usize = 1024;
            if phrase_bytes.len() > MAX_PHRASE_BYTES {
                let _ = event_tx.send(Web3WorkerEvent::Notification(
                    "Error: frase excede longitud maxima de 1KB".to_string(),
                ));
                phrase_bytes.zeroize();
                return;
            }
            match std::str::from_utf8(&phrase_bytes) {
                Ok(phrase) => {
                    if supervisor.breaker.resolve_and_reset(cajas, phrase).is_ok() {
                        let _ = event_tx.send(Web3WorkerEvent::StatusChanged(
                            supervisor.breaker.state.clone(),
                        ));
                        let _ = event_tx.send(Web3WorkerEvent::Notification(
                            "Circuito de seguridad rearmado".to_string(),
                        ));
                    } else {
                        supervisor.breaker.integrity_failures =
                            supervisor.breaker.integrity_failures.saturating_add(1);
                        if supervisor.breaker.integrity_failures >= 3 {
                            supervisor.breaker.trigger_hard_lock();
                            let _ = event_tx.send(Web3WorkerEvent::StatusChanged(
                                supervisor.breaker.state.clone(),
                            ));
                        }
                        let _ = event_tx.send(Web3WorkerEvent::Notification(
                            "Error: credenciales de rearme invalidas".to_string(),
                        ));
                    }
                }
                Err(_) => {
                    let _ = event_tx.send(Web3WorkerEvent::Notification(
                        "Error: frase con codificacion UTF-8 invalida".to_string(),
                    ));
                }
            }
            phrase_bytes.zeroize();
        }
        Web3WorkerCmd::IncomingWebRequest(mut payload, mut stream) => {
            let old_state = supervisor.breaker.state.clone();
            let response = supervisor.handle_native_payload(&payload);
            let _ = write_native_message(&mut stream, &response);
            payload.zeroize();

            if supervisor.breaker.state != old_state {
                let _ = event_tx.send(Web3WorkerEvent::StatusChanged(
                    supervisor.breaker.state.clone(),
                ));
            }
        }
        Web3WorkerCmd::Shutdown => {}
    }
}

fn normalize_id(value: &mut String) {
    *value = value.trim().to_string();
}

fn resolve_socket_path() -> Result<PathBuf, String> {
    let runtime_dir = std::env::var_os("XDG_RUNTIME_DIR").ok_or_else(|| {
        "XDG_RUNTIME_DIR no esta definido; no se abrira socket inseguro en /tmp.".to_string()
    })?;

    let dir = PathBuf::from(runtime_dir).join("livencrypt");
    std::fs::create_dir_all(&dir)
        .map_err(|error| format!("No se pudo crear directorio IPC: {}", error))?;
    std::fs::set_permissions(&dir, std::fs::Permissions::from_mode(0o700))
        .map_err(|error| format!("No se pudo asegurar permisos IPC: {}", error))?;

    Ok(dir.join("livencrypt.sock"))
}

fn remove_existing_socket_safely(socket_path: &PathBuf) -> Result<(), String> {
    if let Ok(meta) = std::fs::symlink_metadata(socket_path) {
        if meta.file_type().is_symlink() || !meta.file_type().is_socket() {
            return Err("Ruta IPC existente no es socket seguro; abortando.".to_string());
        }
        std::fs::remove_file(socket_path)
            .map_err(|error| format!("No se pudo eliminar socket previo: {}", error))?;
    }
    Ok(())
}

#[cfg(unix)]
fn ensure_peer_is_current_user(stream: &UnixStream) -> Result<(), String> {
    use std::mem;
    use std::os::fd::AsRawFd;

    #[cfg(target_os = "linux")]
    {
        let mut cred: libc::ucred = unsafe { mem::zeroed() };
        let mut len = mem::size_of::<libc::ucred>() as libc::socklen_t;

        let rc = unsafe {
            libc::getsockopt(
                stream.as_raw_fd(),
                libc::SOL_SOCKET,
                libc::SO_PEERCRED,
                &mut cred as *mut _ as *mut libc::c_void,
                &mut len,
            )
        };

        if rc != 0 {
            return Err("No se pudo verificar credenciales del peer IPC.".to_string());
        }

        let current_uid = unsafe { libc::geteuid() };
        if cred.uid != current_uid {
            return Err("Conexion IPC rechazada: UID no autorizado.".to_string());
        }

        Ok(())
    }

    #[cfg(not(target_os = "linux"))]
    {
        use std::os::unix::io::AsRawFd;

        let fd = stream.as_raw_fd();
        let mut euid: libc::uid_t = 0;
        let mut egid: libc::gid_t = 0;

        let rc = unsafe { libc::getpeereid(fd, &mut euid, &mut egid) };

        if rc != 0 {
            return Err("No se pudo verificar credenciales del peer IPC.".to_string());
        }

        let current_uid = unsafe { libc::geteuid() };
        if euid != current_uid {
            return Err("Conexion IPC rechazada: UID no autorizado.".to_string());
        }

        Ok(())
    }
}

/// Tareas paralelas ideales para procesar chunks — detectado en startup según CPU
static IDEAL_CHUNK_TASKS: AtomicUsize = AtomicUsize::new(24);
const STREAMING_BINARY_THRESHOLD_BYTES: u64 = 64 * 1024 * 1024;
const FILE_IO_BUFFER_BYTES: usize = 1024 * 1024;
const FILE_V2_MAGIC: [u8; 8] = *b"LVCF2\0\0\0";

#[derive(Serialize, Deserialize, Clone)]
struct FileKdfParams {
    algorithm: String,
    memory_kib: u32,
    iterations: u32,
    parallelism: u32,
}

impl Default for FileKdfParams {
    fn default() -> Self {
        Self {
            algorithm: "argon2id-v19".to_string(),
            memory_kib: 128 * 1024,
            iterations: 3,
            parallelism: 1,
        }
    }
}

#[derive(Serialize, Deserialize, Clone)]
struct FileV2Header {
    magic: [u8; 8],
    version: u16,
    kdf: FileKdfParams,
    salt: Vec<u8>,
    nonce: Vec<u8>,
    template_sha256: Vec<u8>,
    cajas_sha256: Vec<u8>,
    referencia_sha256: Vec<u8>,
    is_txt_output_hint: bool,
}

#[derive(Serialize, Deserialize)]
struct FileV2Envelope {
    header: FileV2Header,
    ciphertext: Vec<u8>,
}

/// Calcular chunk_size adaptativo continuo según tamaño × CPU.
/// Retorna None si el archivo es tan pequeño que conviene secuencial.
#[inline]
fn adaptive_chunk_size(total_chars: usize) -> Option<usize> {
    let tasks = IDEAL_CHUNK_TASKS.load(Ordering::Relaxed);
    let chars_per_task = 2_000_000; // punto de inflexión: 1 tarea más = vale la pena
    let min_parallel = 500_000; // por debajo de esto: secuencial

    // Cuántos chunks ideales: crece con el archivo hasta saturar los cores
    let ideal = (total_chars / chars_per_task).clamp(1, tasks);
    let size = total_chars / ideal;

    if size < min_parallel {
        None
    } else {
        Some(size.clamp(min_parallel, 20_000_000))
    }
}

/// Detectar núcleos CPU una vez al iniciar — llama desde lib.rs::run()
pub fn configure_chunk_tasks() {
    let cpus = std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4);
    let tasks = (cpus * 3).clamp(8, 48);
    IDEAL_CHUNK_TASKS.store(tasks, Ordering::Relaxed);
    println!(
        "[config] {} núcleos CPU → {} tareas paralelas por archivo",
        cpus, tasks
    );
}

fn random_bytes(len: usize) -> Vec<u8> {
    let mut bytes = vec![0u8; len];
    rand::rngs::OsRng.fill_bytes(&mut bytes);
    bytes
}

fn sync_parent_dir(path: &Path) {
    if let Some(parent) = path.parent() {
        if let Ok(dir) = File::open(parent) {
            let _ = dir.sync_all();
        }
    }
}

fn write_file_synced(path: &Path, data: &[u8]) -> Result<(), String> {
    let mut file = OpenOptions::new()
        .write(true)
        .create(true)
        .truncate(true)
        .open(path)
        .map_err(|error| error.to_string())?;
    file.write_all(data).map_err(|error| error.to_string())?;
    file.sync_all().map_err(|error| error.to_string())?;
    drop(file);
    sync_parent_dir(path);
    Ok(())
}

fn sha256_bytes(bytes: &[u8]) -> Vec<u8> {
    Sha256::digest(bytes).to_vec()
}

fn template_sha256(template: &crate::templates::TemplateData) -> Result<Vec<u8>, String> {
    let encoded = rmp_serde::to_vec(template)
        .map_err(|error| format!("No se pudo serializar plantilla para hash: {}", error))?;
    Ok(sha256_bytes(&encoded))
}

fn cajas_sha256(cajas: [i64; 5]) -> Vec<u8> {
    let mut material = Vec::new();
    for caja in cajas {
        material.extend_from_slice(&caja.to_le_bytes());
    }
    let out = sha256_bytes(&material);
    material.zeroize();
    out
}

fn derive_file_key(
    file_secret: &str,
    referencia: &str,
    cajas: [i64; 5],
    salt: &[u8],
    params: &FileKdfParams,
) -> Result<[u8; 32], String> {
    if params.algorithm != "argon2id-v19" {
        return Err(format!("KDF de archivo no soportado: {}", params.algorithm));
    }
    if file_secret.trim().chars().count() < 12 {
        return Err("La clave/frase de archivo debe tener al menos 12 caracteres.".to_string());
    }

    let argon_params = Params::new(
        params.memory_kib,
        params.iterations,
        params.parallelism,
        Some(32),
    )
    .map_err(|error| format!("Parametros Argon2 invalidos: {}", error))?;
    let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, argon_params);

    let mut material = Vec::new();
    material.extend_from_slice(b"LIVENCRYPT-FILE-V2|");
    material.extend_from_slice(referencia.as_bytes());
    material.push(b'|');
    for caja in cajas {
        material.extend_from_slice(&caja.to_le_bytes());
    }
    material.push(b'|');
    material.extend_from_slice(file_secret.as_bytes());

    let mut key = [0u8; 32];
    argon2
        .hash_password_into(&material, salt, &mut key)
        .map_err(|error| format!("No se pudo derivar clave de archivo: {}", error))?;
    material.zeroize();
    Ok(key)
}

fn derive_xor_seed(file_secret: &str, referencia: &str, cajas: [i64; 5]) -> [u8; 32] {
    // CWE-1204/CWE-329: una semilla o IV fijo permite correlacionar mensajes y
    // facilita ataques de diccionario. Esta semilla interna se deriva del secreto
    // de archivo y metadatos; el contenedor AEAD usa nonce aleatorio por cifrado.
    let mut material = Vec::new();
    material.extend_from_slice(b"LIVENCRYPT-FILE-INNER-SEED-V1|");
    material.extend_from_slice(file_secret.as_bytes());
    material.push(b'|');
    material.extend_from_slice(referencia.as_bytes());
    material.push(b'|');
    for caja in cajas {
        material.extend_from_slice(&caja.to_le_bytes());
    }

    let hash = Sha256::digest(&material);
    let mut seed = [0u8; 32];
    seed.copy_from_slice(&hash[..32]);
    material.zeroize();
    seed
}

fn seal_v2(
    plaintext: &[u8],
    header: FileV2Header,
    key: &[u8; 32],
) -> Result<FileV2Envelope, String> {
    if header.nonce.len() != 24 {
        return Err("Nonce AEAD invalido.".to_string());
    }
    let aad = rmp_serde::to_vec(&header)
        .map_err(|error| format!("No se pudo serializar AAD: {}", error))?;
    let cipher = XChaCha20Poly1305::new(Key::from_slice(key));
    let nonce = XNonce::from_slice(&header.nonce);
    let ciphertext = cipher
        .encrypt(
            nonce,
            Payload {
                msg: plaintext,
                aad: &aad,
            },
        )
        .map_err(|_| "Fallo AEAD al cifrar archivo.".to_string())?;
    Ok(FileV2Envelope { header, ciphertext })
}

fn open_v2(envelope: &FileV2Envelope, key: &[u8; 32]) -> Result<Vec<u8>, String> {
    if envelope.header.magic != FILE_V2_MAGIC || envelope.header.version != 2 {
        return Err("Formato Livencrypt de archivo no soportado.".to_string());
    }
    if envelope.header.nonce.len() != 24 {
        return Err("Nonce AEAD invalido.".to_string());
    }
    let aad = rmp_serde::to_vec(&envelope.header)
        .map_err(|error| format!("No se pudo serializar AAD: {}", error))?;
    let cipher = XChaCha20Poly1305::new(Key::from_slice(key));
    let nonce = XNonce::from_slice(&envelope.header.nonce);
    cipher
        .decrypt(
            nonce,
            Payload {
                msg: envelope.ciphertext.as_ref(),
                aad: &aad,
            },
        )
        .map_err(|_| {
            "Autenticacion AEAD fallida: archivo manipulado o clave incorrecta.".to_string()
        })
}

fn verify_file_readable(path: &Path) -> Result<(), String> {
    let mut file = File::open(path).map_err(|error| error.to_string())?;
    let mut buffer = [0u8; 64 * 1024];
    loop {
        let read = file.read(&mut buffer).map_err(|error| error.to_string())?;
        if read == 0 {
            break;
        }
        buffer[..read].zeroize();
    }
    Ok(())
}

fn promote_temp_file(temp_path: &Path, output_path: &Path) -> Result<(), String> {
    verify_file_readable(temp_path)?;
    std::fs::rename(temp_path, output_path).map_err(|error| error.to_string())?;
    sync_parent_dir(output_path);
    Ok(())
}

fn overwrite_and_remove_file(path: &Path) -> Result<(), String> {
    let metadata = match std::fs::metadata(path) {
        Ok(metadata) => metadata,
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(()),
        Err(error) => return Err(error.to_string()),
    };

    if !metadata.is_file() {
        return Ok(());
    }

    let mut file = OpenOptions::new()
        .write(true)
        .open(path)
        .map_err(|error| error.to_string())?;
    let mut remaining = metadata.len().max(1);
    while remaining > 0 {
        let chunk_len = remaining.min(64 * 1024) as usize;
        file.write_all(&random_bytes(chunk_len))
            .map_err(|error| error.to_string())?;
        remaining = remaining.saturating_sub(chunk_len as u64);
    }
    file.sync_all().map_err(|error| error.to_string())?;
    drop(file);
    std::fs::remove_file(path).map_err(|error| error.to_string())?;
    sync_parent_dir(path);
    Ok(())
}

fn cleanup_temp_file(path: &Path) {
    let _ = overwrite_and_remove_file(path);
}

pub struct FileCipher;

impl FileCipher {
    pub fn encrypt_file_authenticated(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        template: &crate::templates::TemplateData,
        cajas: [i64; 5],
        file_secret: &str,
    ) -> Result<(), String> {
        let inner_path = temp_inner_path(output_path);
        cleanup_temp_file(&inner_path);

        let mut xor_seed = derive_xor_seed(file_secret, referencia, cajas);
        let inner_result = Self::encrypt_file(
            input_path,
            &inner_path,
            referencia,
            template,
            cajas[0],
            cajas[1],
            cajas[2],
            cajas[3],
            cajas[4],
            &xor_seed,
        );
        xor_seed.zeroize();
        if let Err(error) = inner_result {
            cleanup_temp_file(&inner_path);
            return Err(error);
        }

        let mut inner = std::fs::read(&inner_path)
            .map_err(|error| format!("No se pudo leer payload interno: {}", error))?;
        cleanup_temp_file(&inner_path);

        let kdf = FileKdfParams::default();
        let salt = random_bytes(32);
        let nonce = random_bytes(24);
        // El nonce AEAD debe ser unico e impredecible por archivo; reutilizarlo
        // con la misma clave comprometeria la confidencialidad del ciphertext.
        let mut key = derive_file_key(file_secret, referencia, cajas, &salt, &kdf)?;

        let header = FileV2Header {
            magic: FILE_V2_MAGIC,
            version: 2,
            kdf,
            salt,
            nonce,
            template_sha256: template_sha256(template)?,
            cajas_sha256: cajas_sha256(cajas),
            referencia_sha256: sha256_bytes(referencia.as_bytes()),
            is_txt_output_hint: false,
        };

        let envelope = seal_v2(&inner, header, &key);
        inner.zeroize();
        key.zeroize();

        let envelope = envelope?;
        let encoded = rmp_serde::to_vec(&envelope)
            .map_err(|error| format!("No se pudo serializar archivo LVC v2: {}", error))?;
        write_file_synced(output_path, &encoded)
    }

    pub fn decrypt_file_authenticated(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        is_txt_output: bool,
        template: &crate::templates::TemplateData,
        cajas: [i64; 5],
        file_secret: &str,
    ) -> Result<(), String> {
        let bytes = std::fs::read(input_path)
            .map_err(|error| format!("No se pudo leer archivo cifrado: {}", error))?;
        let envelope: FileV2Envelope = rmp_serde::from_slice(&bytes)
            .map_err(|error| format!("Archivo LVC v2 invalido: {}", error))?;

        if envelope.header.template_sha256 != template_sha256(template)? {
            return Err("La plantilla no coincide con la usada al cifrar.".to_string());
        }
        if envelope.header.cajas_sha256 != cajas_sha256(cajas) {
            return Err("Las cajas no coinciden con las usadas al cifrar.".to_string());
        }
        if envelope.header.referencia_sha256 != sha256_bytes(referencia.as_bytes()) {
            return Err("La referencia no coincide con la usada al cifrar.".to_string());
        }

        let mut key = derive_file_key(
            file_secret,
            referencia,
            cajas,
            &envelope.header.salt,
            &envelope.header.kdf,
        )?;
        let mut inner = open_v2(&envelope, &key)?;
        key.zeroize();

        let inner_path = temp_inner_path(output_path);
        cleanup_temp_file(&inner_path);
        write_file_synced(&inner_path, &inner)?;
        inner.zeroize();

        let mut xor_seed = derive_xor_seed(file_secret, referencia, cajas);
        let result = Self::decrypt_file(
            &inner_path,
            output_path,
            referencia,
            is_txt_output,
            template,
            cajas[0],
            cajas[1],
            cajas[2],
            cajas[3],
            cajas[4],
            &xor_seed,
        );
        xor_seed.zeroize();
        cleanup_temp_file(&inner_path);
        result
    }

    pub fn encrypt_file(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        template: &crate::templates::TemplateData,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        seed: &[u8; 32],
    ) -> Result<(), String> {
        let t_start = Instant::now();
        let metadata = std::fs::metadata(input_path).map_err(|e| e.to_string())?;
        let is_txt_input = input_path
            .extension()
            .and_then(|e| e.to_str())
            .map(|e| e.to_lowercase() == "txt")
            .unwrap_or(false);

        if !is_txt_input && metadata.len() >= STREAMING_BINARY_THRESHOLD_BYTES {
            return Self::encrypt_file_streaming_binary(
                input_path,
                output_path,
                referencia,
                template,
                caja1,
                caja2,
                caja3,
                caja4,
                caja5,
                seed,
                t_start,
            );
        }

        let mut data = std::fs::read(input_path).map_err(|e| e.to_string())?;
        let mut payload = if is_txt_input {
            let compressed = Self::compress(&data)?;
            data.zeroize();
            compressed
        } else {
            std::mem::take(&mut data)
        };

        // Capa adicional sobre el payload interno serializado a hex.
        // Si seed = [0;32], el XOR es identidad para compatibilidad legacy.
        if seed.iter().any(|&b| b != 0) {
            for (i, byte) in payload.iter_mut().enumerate() {
                *byte ^= seed[i % 32];
            }
        }

        let hex_string = hex::encode(&payload);
        payload.zeroize();

        let grid = template.as_grid();
        let rows = template.rows as usize;
        let cols = template.cols as usize;
        let is_filelivencrypt = rows == 5 && cols == 23;

        let encrypted_hex = if is_filelivencrypt {
            Self::procesar_texto_archivo(
                &hex_string,
                caja1,
                caja2,
                caja3,
                caja4,
                caja5,
                "cifrar",
                &grid,
            )
        } else {
            procesar_texto_archivo_variable(
                &hex_string,
                caja1,
                caja2,
                caja3,
                caja4,
                caja5,
                "cifrar",
                &grid,
                rows,
                cols,
            )
        };

        let sealed = if !referencia.is_empty() {
            Self::insertar_referencia_aleatoria(&encrypted_hex, referencia)
        } else {
            encrypted_hex
        };

        let compressed = Self::compress(sealed.as_bytes())?;
        write_file_synced(output_path, &compressed)?;
        let elapsed = t_start.elapsed().as_millis();
        println!(
            "[cipher] CIFRADO+COMPRESO {} → {} ({}ms, {} hex chars, {}×{})",
            input_path.file_name().unwrap_or_default().to_string_lossy(),
            output_path
                .file_name()
                .unwrap_or_default()
                .to_string_lossy(),
            elapsed,
            sealed.len(),
            rows,
            cols
        );
        Ok(())
    }

    pub fn decrypt_file(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        is_txt_output: bool,
        template: &crate::templates::TemplateData,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        seed: &[u8; 32],
    ) -> Result<(), String> {
        let t_start = Instant::now();
        Self::decrypt_file_streaming(
            input_path,
            output_path,
            referencia,
            is_txt_output,
            template,
            caja1,
            caja2,
            caja3,
            caja4,
            caja5,
            seed,
            t_start,
        )
    }

    pub fn load_file_info(path: &Path) -> Result<FileInfo, String> {
        let metadata = std::fs::metadata(path).map_err(|e| e.to_string())?;
        let name = path
            .file_name()
            .unwrap_or_default()
            .to_string_lossy()
            .to_string();
        let size_bytes = metadata.len();

        let is_text = path
            .extension()
            .and_then(|e| e.to_str())
            .map(|e| e.to_lowercase() == "txt")
            .unwrap_or(false);

        let hex_chars = if is_text {
            let data = std::fs::read(path).map_err(|e| e.to_string())?;
            let compressed = Self::compress(&data)?;
            (compressed.len() * 2) as u64
        } else {
            size_bytes * 2
        };

        Ok(FileInfo {
            name,
            size_bytes,
            hex_chars,
            is_compressed: is_text,
        })
    }

    fn encrypt_file_streaming_binary(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        template: &crate::templates::TemplateData,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        seed: &[u8; 32],
        t_start: Instant,
    ) -> Result<(), String> {
        let input_len = std::fs::metadata(input_path)
            .map_err(|e| e.to_string())?
            .len();
        let total_chars_u64 = input_len
            .checked_mul(2)
            .ok_or("Archivo demasiado grande para indexado hex")?;
        let total_chars = usize::try_from(total_chars_u64)
            .map_err(|_| "Archivo demasiado grande para esta arquitectura")?;

        let grid = template.as_grid();
        let rows = template.rows as usize;
        let cols = template.cols as usize;
        if rows == 0 || cols == 0 {
            return Err("Plantilla de archivo vacia".into());
        }
        let char_map = build_char_map(&grid);
        let ref_pos = if referencia.is_empty() {
            None
        } else {
            Some(rand::thread_rng().gen_range(0..=total_chars))
        };

        let temp_path = temp_output_path(output_path);
        cleanup_temp_file(&temp_path);
        let write_result = (|| -> Result<(), String> {
            let input = File::open(input_path).map_err(|e| e.to_string())?;
            let output = File::create(&temp_path).map_err(|e| e.to_string())?;
            let mut reader = BufReader::with_capacity(FILE_IO_BUFFER_BYTES, input);
            let writer = BufWriter::with_capacity(FILE_IO_BUFFER_BYTES, output);
            let mut encoder =
                zstd::stream::write::Encoder::new(writer, 3).map_err(|e| e.to_string())?;

            let mut buffer = vec![0u8; FILE_IO_BUFFER_BYTES];
            let mut byte_offset = 0usize;
            let mut char_index = 0usize;
            let use_seed = seed.iter().any(|&b| b != 0);
            const HEX: &[u8; 16] = b"0123456789abcdef";

            loop {
                let n = reader.read(&mut buffer).map_err(|e| e.to_string())?;
                if n == 0 {
                    break;
                }
                for (local_i, &raw) in buffer[..n].iter().enumerate() {
                    let absolute_byte = byte_offset + local_i;
                    let byte = if use_seed {
                        raw ^ seed[absolute_byte % 32]
                    } else {
                        raw
                    };
                    for nibble in [byte >> 4, byte & 0x0f] {
                        if ref_pos == Some(char_index) {
                            encoder
                                .write_all(referencia.as_bytes())
                                .map_err(|e| e.to_string())?;
                        }
                        let hex_char = HEX[nibble as usize] as char;
                        let encrypted = transform_file_char(
                            hex_char,
                            char_index,
                            total_chars,
                            caja1,
                            caja2,
                            caja3,
                            caja4,
                            caja5,
                            true,
                            &grid,
                            &char_map,
                            rows,
                            cols,
                        )?;
                        let mut out = [0u8; 4];
                        encoder
                            .write_all(encrypted.encode_utf8(&mut out).as_bytes())
                            .map_err(|e| e.to_string())?;
                        char_index += 1;
                    }
                }
                byte_offset += n;
                buffer[..n].zeroize();
            }

            if ref_pos == Some(total_chars) {
                encoder
                    .write_all(referencia.as_bytes())
                    .map_err(|e| e.to_string())?;
            }
            let mut writer = encoder.finish().map_err(|e| e.to_string())?;
            writer.flush().map_err(|e| e.to_string())?;
            writer.get_ref().sync_all().map_err(|e| e.to_string())?;
            Ok(())
        })();

        if let Err(e) = write_result {
            cleanup_temp_file(&temp_path);
            return Err(e);
        }

        if let Err(e) = promote_temp_file(&temp_path, output_path) {
            cleanup_temp_file(&temp_path);
            return Err(e);
        }

        let elapsed = t_start.elapsed().as_millis();
        println!(
            "[cipher] CIFRADO+COMPRESO streaming {} → {} ({}ms, {} hex chars, {}×{})",
            input_path.file_name().unwrap_or_default().to_string_lossy(),
            output_path
                .file_name()
                .unwrap_or_default()
                .to_string_lossy(),
            elapsed,
            total_chars,
            rows,
            cols
        );
        Ok(())
    }

    fn decrypt_file_streaming(
        input_path: &Path,
        output_path: &Path,
        referencia: &str,
        is_txt_output: bool,
        template: &crate::templates::TemplateData,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        seed: &[u8; 32],
        t_start: Instant,
    ) -> Result<(), String> {
        let grid = template.as_grid();
        let rows = template.rows as usize;
        let cols = template.cols as usize;
        if rows == 0 || cols == 0 {
            return Err("Plantilla de archivo vacia".into());
        }
        let char_map = build_char_map(&grid);
        let total_chars = count_clean_encrypted_chars(input_path, referencia)?;
        if total_chars % 2 != 0 {
            return Err("Hex descifrado con longitud impar".into());
        }

        let temp_path = temp_output_path(output_path);
        let decoded_temp_path = temp_path.with_extension("decoded.tmp");
        cleanup_temp_file(&temp_path);
        cleanup_temp_file(&decoded_temp_path);

        let write_result = (|| -> Result<(), String> {
            let decoded_output = File::create(&decoded_temp_path).map_err(|e| e.to_string())?;
            let mut decoded_writer = BufWriter::with_capacity(FILE_IO_BUFFER_BYTES, decoded_output);
            let use_seed = seed.iter().any(|&b| b != 0);
            let mut char_index = 0usize;
            let mut byte_index = 0usize;
            let mut high_nibble: Option<u8> = None;

            process_clean_encrypted_chars(input_path, referencia, |encrypted| {
                let hex_char = transform_file_char(
                    encrypted,
                    char_index,
                    total_chars,
                    caja1,
                    caja2,
                    caja3,
                    caja4,
                    caja5,
                    false,
                    &grid,
                    &char_map,
                    rows,
                    cols,
                )?;
                let nibble = hex_char
                    .to_digit(16)
                    .ok_or_else(|| format!("Hex invalido tras descifrar: {}", hex_char))?
                    as u8;
                if let Some(high) = high_nibble.take() {
                    let mut byte = (high << 4) | nibble;
                    if use_seed {
                        byte ^= seed[byte_index % 32];
                    }
                    decoded_writer
                        .write_all(&[byte])
                        .map_err(|e| e.to_string())?;
                    byte.zeroize();
                    byte_index += 1;
                } else {
                    high_nibble = Some(nibble);
                }
                char_index += 1;
                Ok(())
            })?;

            if high_nibble.is_some() {
                return Err("Hex descifrado con longitud impar".into());
            }
            decoded_writer.flush().map_err(|e| e.to_string())?;
            decoded_writer
                .get_ref()
                .sync_all()
                .map_err(|e| e.to_string())?;

            if is_txt_output {
                let decoded_input = File::open(&decoded_temp_path).map_err(|e| e.to_string())?;
                let mut temp_output = File::create(&temp_path).map_err(|e| e.to_string())?;
                match zstd::stream::copy_decode(decoded_input, &mut temp_output) {
                    Ok(_) => {
                        temp_output.sync_all().map_err(|e| e.to_string())?;
                        drop(temp_output);
                    }
                    Err(_) => {
                        drop(temp_output);
                        let mut decoded =
                            std::fs::read(&decoded_temp_path).map_err(|e| e.to_string())?;
                        write_file_synced(&temp_path, &decoded)?;
                        decoded.zeroize();
                    }
                }
                verify_file_readable(&temp_path)?;
            } else {
                promote_temp_file(&decoded_temp_path, &temp_path)?;
            }
            Ok(())
        })();

        if let Err(e) = write_result {
            cleanup_temp_file(&decoded_temp_path);
            cleanup_temp_file(&temp_path);
            return Err(e);
        }

        if let Err(e) = promote_temp_file(&temp_path, output_path) {
            cleanup_temp_file(&decoded_temp_path);
            cleanup_temp_file(&temp_path);
            return Err(e);
        }
        cleanup_temp_file(&decoded_temp_path);

        let elapsed = t_start.elapsed().as_millis();
        println!(
            "[cipher] DESCOMPRESO+DESCIFRADO streaming {} → {} ({}ms, {}×{})",
            input_path.file_name().unwrap_or_default().to_string_lossy(),
            output_path
                .file_name()
                .unwrap_or_default()
                .to_string_lossy(),
            elapsed,
            rows,
            cols
        );
        Ok(())
    }

    pub fn save_result(temp_path: &Path, save_path: &Path) -> Result<(), String> {
        std::fs::copy(temp_path, save_path).map_err(|e| e.to_string())?;
        overwrite_and_remove_file(temp_path)?;
        Ok(())
    }
}

#[derive(Serialize, Clone)]
pub struct FileInfo {
    pub name: String,
    pub size_bytes: u64,
    pub hex_chars: u64,
    pub is_compressed: bool,
}

impl FileCipher {
    // SAFETY: from_utf8_unchecked es seguro porque los bytes provienen de
    // char::encode_utf8() sobre chars de la grid, todos Unicode válidos.
    #[allow(unsafe_code)]
    pub fn procesar_texto_archivo(
        hex_text: &str,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        mode: &str,
        grid: &[Vec<char>],
    ) -> String {
        let chars: Vec<char> = hex_text.chars().collect();
        let total_chars = chars.len();
        let char_map = build_char_map(grid);

        // filelivencrypt grid is always 5 rows × 23 cols — constants enable LLVM strength reduction
        let is_cifrar = mode == "cifrar";

        // Decidir secuencial vs paralelo con chunk_size adaptativo continuo
        match adaptive_chunk_size(total_chars) {
            None => {
                // Secuencial: archivo demasiado pequeño para justificar rayon
                let mut out = Vec::with_capacity(total_chars + total_chars / 4);
                for (i, &c) in chars.iter().enumerate() {
                    if let Some(&(mut fila, mut columna)) = char_map.get(c) {
                        let indice: i64 = if is_cifrar {
                            (total_chars - i) as i64
                        } else {
                            (i + 1) as i64
                        };
                        let mut fc = columna as i64;
                        let mut fr = fila as i64;
                        if is_cifrar {
                            fc = (fc + caja1 - indice).rem_euclid(23);
                            fr = (fr + caja2 - indice).rem_euclid(5);
                            fc = (fc - caja3 + indice).rem_euclid(23);
                            fr = (fr - caja4 + indice).rem_euclid(5);
                        } else {
                            fr = (fr + caja4 - indice).rem_euclid(5);
                            fc = (fc + caja3 - indice).rem_euclid(23);
                            fr = (fr - caja2 + indice).rem_euclid(5);
                            fc = (fc - caja1 + indice).rem_euclid(23);
                        }
                        fila = fr as usize;
                        columna = fc as usize;
                        let movs = get_caja5_movements_inline(i as i64, caja5, total_chars);
                        for (mov_fila, mov_columna) in movs {
                            if is_cifrar {
                                fila = (fila as i64 + mov_fila).rem_euclid(5) as usize;
                                columna = (columna as i64 + mov_columna).rem_euclid(23) as usize;
                            } else {
                                fila = (fila as i64 - mov_fila).rem_euclid(5) as usize;
                                columna = (columna as i64 - mov_columna).rem_euclid(23) as usize;
                            }
                        }
                        let mut buf = [0u8; 4];
                        let encoded = grid[fila][columna].encode_utf8(&mut buf);
                        out.extend_from_slice(encoded.as_bytes());
                    }
                }
                // SAFETY: out contiene bytes de char::encode_utf8() sobre chars
                // de la grid, todos Unicode válidos. La concatenación de segmentos
                // UTF-8 válidos preserva UTF-8.
                return unsafe { String::from_utf8_unchecked(out) };
            }
            Some(chunk_size) => {
                // Paralelo con rayon: map produce Vec<u8> por chunk, merge secuencial al final
                let chunks: Vec<Vec<u8>> = chars
                    .par_chunks(chunk_size)
                    .enumerate()
                    .map(|(chunk_idx, chunk)| {
                        let offset = chunk_idx * chunk_size;
                        let mut out = Vec::with_capacity(chunk.len() + chunk.len() / 4);
                        for (local_i, &c) in chunk.iter().enumerate() {
                            let i = offset + local_i;
                            if let Some(&(mut fila, mut columna)) = char_map.get(c) {
                                let indice: i64 = if is_cifrar {
                                    (total_chars - i) as i64
                                } else {
                                    (i + 1) as i64
                                };
                                let mut fc = columna as i64;
                                let mut fr = fila as i64;

                                if is_cifrar {
                                    fc = (fc + caja1 - indice).rem_euclid(23);
                                    fr = (fr + caja2 - indice).rem_euclid(5);
                                    fc = (fc - caja3 + indice).rem_euclid(23);
                                    fr = (fr - caja4 + indice).rem_euclid(5);
                                } else {
                                    fr = (fr + caja4 - indice).rem_euclid(5);
                                    fc = (fc + caja3 - indice).rem_euclid(23);
                                    fr = (fr - caja2 + indice).rem_euclid(5);
                                    fc = (fc - caja1 + indice).rem_euclid(23);
                                }

                                fila = fr as usize;
                                columna = fc as usize;

                                let movs = get_caja5_movements_inline(i as i64, caja5, total_chars);
                                for (mov_fila, mov_columna) in movs {
                                    if is_cifrar {
                                        fila = (fila as i64 + mov_fila).rem_euclid(5) as usize;
                                        columna =
                                            (columna as i64 + mov_columna).rem_euclid(23) as usize;
                                    } else {
                                        fila = (fila as i64 - mov_fila).rem_euclid(5) as usize;
                                        columna =
                                            (columna as i64 - mov_columna).rem_euclid(23) as usize;
                                    }
                                }

                                let mut buf = [0u8; 4];
                                let encoded = grid[fila][columna].encode_utf8(&mut buf);
                                out.extend_from_slice(encoded.as_bytes());
                            }
                        }
                        out
                    })
                    .collect();

                // Merge secuencial — una sola pasada, sin merges en árbol binario
                let total_bytes: usize = chunks.iter().map(|v| v.len()).sum();
                let mut resultado = Vec::with_capacity(total_bytes);
                for chunk in chunks {
                    resultado.extend_from_slice(&chunk);
                }
                // SAFETY: cada chunk contiene solo bytes de encode_utf8(),
                // la concatenación preserva UTF-8.
                unsafe { String::from_utf8_unchecked(resultado) }
            }
        }
    }
}

/// Dual-lookup: array[128] para ASCII (1ns, sin hashing) + HashMap para Unicode
struct CharLookup {
    ascii: [Option<(usize, usize)>; 128],
    unicode: HashMap<char, (usize, usize)>,
}

impl CharLookup {
    #[inline]
    fn get(&self, c: char) -> Option<&(usize, usize)> {
        let byte = c as u32;
        if byte < 128 {
            self.ascii[byte as usize].as_ref()
        } else {
            self.unicode.get(&c)
        }
    }
}

/// Build dual lookup: array (ASCII hex) + HashMap (Unicode grid chars)
fn build_char_map(grid: &[Vec<char>]) -> CharLookup {
    let mut ascii: [Option<(usize, usize)>; 128] = [None; 128];
    let mut unicode = HashMap::with_capacity(grid.iter().map(|r| r.len()).sum());

    for (r, row) in grid.iter().enumerate() {
        for (c, &cell) in row.iter().enumerate() {
            let lower = cell.to_lowercase().next().unwrap_or(cell);
            let upper = cell.to_uppercase().next().unwrap_or(cell);

            // ASCII chars → array (fast path)
            for &ch in &[lower, upper] {
                let byte = ch as u32;
                if byte < 128 {
                    ascii[byte as usize] = Some((r, c));
                } else {
                    unicode.entry(ch).or_insert((r, c));
                }
            }
        }
    }
    CharLookup { ascii, unicode }
}

fn temp_output_path(output_path: &Path) -> PathBuf {
    let extension = output_path
        .extension()
        .and_then(|ext| ext.to_str())
        .map(|ext| format!("{}.tmp", ext))
        .unwrap_or_else(|| "tmp".to_string());
    output_path.with_extension(extension)
}

fn temp_inner_path(output_path: &Path) -> PathBuf {
    output_path.with_extension(format!("inner-{}.tmp", std::process::id()))
}

fn count_clean_encrypted_chars(input_path: &Path, referencia: &str) -> Result<usize, String> {
    let mut count = 0usize;
    process_clean_encrypted_chars(input_path, referencia, |_| {
        count = count.saturating_add(1);
        Ok(())
    })?;
    Ok(count)
}

fn process_clean_encrypted_chars<F>(
    input_path: &Path,
    referencia: &str,
    mut on_char: F,
) -> Result<(), String>
where
    F: FnMut(char) -> Result<(), String>,
{
    let input = File::open(input_path).map_err(|e| e.to_string())?;
    let mut decoder = zstd::stream::read::Decoder::new(input).map_err(|e| e.to_string())?;
    let reference: Vec<char> = referencia.chars().collect();
    let has_reference = !reference.is_empty();
    let mut matcher = ReferenceSkipper::new(reference);
    let mut bytes = vec![0u8; FILE_IO_BUFFER_BYTES];
    let mut pending = Vec::<u8>::new();

    loop {
        let n = decoder.read(&mut bytes).map_err(|e| e.to_string())?;
        if n == 0 {
            break;
        }
        pending.extend_from_slice(&bytes[..n]);
        consume_valid_utf8_prefix(&mut pending, |ch| {
            if !has_reference && (ch == '\n' || ch == ' ') {
                return Ok(());
            }
            matcher.push(ch, &mut on_char)
        })?;
    }

    if !pending.is_empty() {
        consume_valid_utf8_prefix(&mut pending, |ch| {
            if !has_reference && (ch == '\n' || ch == ' ') {
                return Ok(());
            }
            matcher.push(ch, &mut on_char)
        })?;
        if !pending.is_empty() {
            return Err("Datos cifrados corruptos: UTF-8 truncado".into());
        }
    }
    matcher.finish(&mut on_char)
}

fn consume_valid_utf8_prefix<F>(bytes: &mut Vec<u8>, mut on_char: F) -> Result<(), String>
where
    F: FnMut(char) -> Result<(), String>,
{
    let valid_len = match std::str::from_utf8(bytes) {
        Ok(text) => {
            for ch in text.chars() {
                on_char(ch)?;
            }
            bytes.clear();
            return Ok(());
        }
        Err(e) => {
            if e.error_len().is_some() {
                return Err("Datos cifrados corruptos: UTF-8 invalido".into());
            }
            e.valid_up_to()
        }
    };

    if valid_len > 0 {
        let text = std::str::from_utf8(&bytes[..valid_len])
            .map_err(|_| "Datos cifrados corruptos: UTF-8 invalido".to_string())?;
        for ch in text.chars() {
            on_char(ch)?;
        }
        bytes.drain(..valid_len);
    }
    Ok(())
}

struct ReferenceSkipper {
    reference: Vec<char>,
    pending: Vec<char>,
    skipped: bool,
}

impl ReferenceSkipper {
    fn new(reference: Vec<char>) -> Self {
        Self {
            reference,
            pending: Vec::new(),
            skipped: false,
        }
    }

    fn push<F>(&mut self, ch: char, emit: &mut F) -> Result<(), String>
    where
        F: FnMut(char) -> Result<(), String>,
    {
        if self.skipped || self.reference.is_empty() {
            return emit(ch);
        }

        self.pending.push(ch);
        loop {
            if self.pending == self.reference {
                self.pending.clear();
                self.skipped = true;
                return Ok(());
            }
            if self.reference.starts_with(&self.pending) {
                return Ok(());
            }
            let first = self.pending.remove(0);
            emit(first)?;
            if self.pending.is_empty() {
                return Ok(());
            }
        }
    }

    fn finish<F>(&mut self, emit: &mut F) -> Result<(), String>
    where
        F: FnMut(char) -> Result<(), String>,
    {
        for ch in self.pending.drain(..) {
            emit(ch)?;
        }
        Ok(())
    }
}

#[allow(clippy::too_many_arguments)]
fn transform_file_char(
    c: char,
    i: usize,
    total_chars: usize,
    caja1: i64,
    caja2: i64,
    caja3: i64,
    caja4: i64,
    caja5: i64,
    is_cifrar: bool,
    grid: &[Vec<char>],
    char_map: &CharLookup,
    rows: usize,
    cols: usize,
) -> Result<char, String> {
    let Some(&(mut fila, mut columna)) = char_map.get(c) else {
        return Err(format!(
            "caracter fuera de grid para cifrado de archivo: {}",
            c
        ));
    };
    let rows_i = rows as i64;
    let cols_i = cols as i64;
    let indice: i64 = if is_cifrar {
        (total_chars - i) as i64
    } else {
        (i + 1) as i64
    };
    let mut fc = columna as i64;
    let mut fr = fila as i64;
    if is_cifrar {
        fc = (fc + caja1 - indice).rem_euclid(cols_i);
        fr = (fr + caja2 - indice).rem_euclid(rows_i);
        fc = (fc - caja3 + indice).rem_euclid(cols_i);
        fr = (fr - caja4 + indice).rem_euclid(rows_i);
    } else {
        fr = (fr + caja4 - indice).rem_euclid(rows_i);
        fc = (fc + caja3 - indice).rem_euclid(cols_i);
        fr = (fr - caja2 + indice).rem_euclid(rows_i);
        fc = (fc - caja1 + indice).rem_euclid(cols_i);
    }
    fila = fr as usize;
    columna = fc as usize;
    let movs = get_caja5_movements_inline(i as i64, caja5, total_chars);
    for (mov_fila, mov_columna) in movs {
        if is_cifrar {
            fila = (fila as i64 + mov_fila).rem_euclid(rows_i) as usize;
            columna = (columna as i64 + mov_columna).rem_euclid(cols_i) as usize;
        } else {
            fila = (fila as i64 - mov_fila).rem_euclid(rows_i) as usize;
            columna = (columna as i64 - mov_columna).rem_euclid(cols_i) as usize;
        }
    }
    Ok(grid[fila][columna])
}

/// Variable path: cualquier plantilla personalizada (N×M) — sin optimización LLVM
#[allow(unsafe_code)]
fn procesar_texto_archivo_variable(
    hex_text: &str,
    caja1: i64,
    caja2: i64,
    caja3: i64,
    caja4: i64,
    caja5: i64,
    mode: &str,
    grid: &[Vec<char>],
    rows: usize,
    cols: usize,
) -> String {
    let chars: Vec<char> = hex_text.chars().collect();
    let total_chars = chars.len();
    let char_map = build_char_map(grid);

    let rows_i = rows as i64;
    let cols_i = cols as i64;
    let is_cifrar = mode == "cifrar";

    match adaptive_chunk_size(total_chars) {
        None => {
            let mut out = Vec::with_capacity(total_chars + total_chars / 4);
            for (i, &c) in chars.iter().enumerate() {
                if let Some(&(mut fila, mut columna)) = char_map.get(c) {
                    let indice: i64 = if is_cifrar {
                        (total_chars - i) as i64
                    } else {
                        (i + 1) as i64
                    };
                    let mut fc = columna as i64;
                    let mut fr = fila as i64;
                    if is_cifrar {
                        fc = (fc + caja1 - indice).rem_euclid(cols_i);
                        fr = (fr + caja2 - indice).rem_euclid(rows_i);
                        fc = (fc - caja3 + indice).rem_euclid(cols_i);
                        fr = (fr - caja4 + indice).rem_euclid(rows_i);
                    } else {
                        fr = (fr + caja4 - indice).rem_euclid(rows_i);
                        fc = (fc + caja3 - indice).rem_euclid(cols_i);
                        fr = (fr - caja2 + indice).rem_euclid(rows_i);
                        fc = (fc - caja1 + indice).rem_euclid(cols_i);
                    }
                    fila = fr as usize;
                    columna = fc as usize;
                    let movs = get_caja5_movements_inline(i as i64, caja5, total_chars);
                    for (mov_fila, mov_columna) in movs {
                        if is_cifrar {
                            fila = (fila as i64 + mov_fila).rem_euclid(rows_i) as usize;
                            columna = (columna as i64 + mov_columna).rem_euclid(cols_i) as usize;
                        } else {
                            fila = (fila as i64 - mov_fila).rem_euclid(rows_i) as usize;
                            columna = (columna as i64 - mov_columna).rem_euclid(cols_i) as usize;
                        }
                    }
                    let mut buf = [0u8; 4];
                    let encoded = grid[fila][columna].encode_utf8(&mut buf);
                    out.extend_from_slice(encoded.as_bytes());
                }
            }
            // SAFETY: out contiene bytes de encode_utf8() de chars de grid.
            // Cada carácter Unicode codificado en UTF-8 es válido.
            return unsafe { String::from_utf8_unchecked(out) };
        }
        Some(chunk_size) => {
            let chunks: Vec<Vec<u8>> = chars
                .par_chunks(chunk_size)
                .enumerate()
                .map(|(chunk_idx, chunk)| {
                    let offset = chunk_idx * chunk_size;
                    let mut out = Vec::with_capacity(chunk.len() + chunk.len() / 4);
                    for (local_i, &c) in chunk.iter().enumerate() {
                        let i = offset + local_i;
                        if let Some(&(mut fila, mut columna)) = char_map.get(c) {
                            let indice: i64 = if is_cifrar {
                                (total_chars - i) as i64
                            } else {
                                (i + 1) as i64
                            };
                            let mut fc = columna as i64;
                            let mut fr = fila as i64;
                            if is_cifrar {
                                fc = (fc + caja1 - indice).rem_euclid(cols_i);
                                fr = (fr + caja2 - indice).rem_euclid(rows_i);
                                fc = (fc - caja3 + indice).rem_euclid(cols_i);
                                fr = (fr - caja4 + indice).rem_euclid(rows_i);
                            } else {
                                fr = (fr + caja4 - indice).rem_euclid(rows_i);
                                fc = (fc + caja3 - indice).rem_euclid(cols_i);
                                fr = (fr - caja2 + indice).rem_euclid(rows_i);
                                fc = (fc - caja1 + indice).rem_euclid(cols_i);
                            }
                            fila = fr as usize;
                            columna = fc as usize;
                            let movs = get_caja5_movements_inline(i as i64, caja5, total_chars);
                            for (mov_fila, mov_columna) in movs {
                                if is_cifrar {
                                    fila = (fila as i64 + mov_fila).rem_euclid(rows_i) as usize;
                                    columna =
                                        (columna as i64 + mov_columna).rem_euclid(cols_i) as usize;
                                } else {
                                    fila = (fila as i64 - mov_fila).rem_euclid(rows_i) as usize;
                                    columna =
                                        (columna as i64 - mov_columna).rem_euclid(cols_i) as usize;
                                }
                            }
                            let mut buf = [0u8; 4];
                            let encoded = grid[fila][columna].encode_utf8(&mut buf);
                            out.extend_from_slice(encoded.as_bytes());
                        }
                    }
                    out
                })
                .collect();
            let total_bytes: usize = chunks.iter().map(|v| v.len()).sum();
            let mut resultado = Vec::with_capacity(total_bytes);
            for chunk in chunks {
                resultado.extend_from_slice(&chunk);
            }
            // SAFETY: chunks generados con encode_utf8() → concatenación UTF-8 válida.
            unsafe { String::from_utf8_unchecked(resultado) }
        }
    }
}

/// Inline caja5 movements — odd indices keep Livencrypt 1.1, even indices use the 2.0 mirror.
#[inline(always)]
fn get_caja5_movements_inline(i: i64, caja5: i64, total_chars: usize) -> &'static [(i64, i64)] {
    if caja5 == (total_chars as i64) - 1 {
        return &[];
    }

    match (i.rem_euclid(2), caja5.rem_euclid(4)) {
        (1, 0) => &[(2, 0), (0, 1)],
        (1, 1) => &[(0, 2), (1, 0)],
        (1, 2) => &[(-2, 0), (0, -1)],
        (1, _) => &[(0, -2), (-1, 0)],
        (0, 0) => &[(-2, 0), (0, -1)],
        (0, 1) => &[(0, -2), (-1, 0)],
        (0, 2) => &[(2, 0), (0, 1)],
        _ => &[(0, 2), (1, 0)],
    }
}

    fn insertar_referencia_aleatoria(content: &str, referencia: &str) -> String {
        let mut rng = rand::thread_rng();
        let char_count = content.chars().count();
        let char_pos = rng.gen_range(0..=char_count);
        let byte_pos = content
            .char_indices()
            .map(|(i, _)| i)
            .chain(std::iter::once(content.len()))
            .nth(char_pos)
            .unwrap_or(0);
        let mut result = String::with_capacity(content.len() + referencia.len());
        result.push_str(&content[..byte_pos]);
        result.push_str(referencia);
        result.push_str(&content[byte_pos..]);
        result
    }

    }

pub fn normalize_control_symbols(symbols: &str) -> String {
    let mut normalized = String::new();
    for symbol in symbols.chars().filter(|symbol| !symbol.is_whitespace()) {
        if !normalized.contains(symbol) {
            normalized.push(symbol);
        }
    }

    if normalized.is_empty() {
        DEFAULT_CONTROL_SYMBOLS.to_string()
    } else {
        normalized
    }
}

pub fn normalize_password_direction(direction: &str) -> String {
    match direction {
        PASSWORD_DIRECTION_AFTER => PASSWORD_DIRECTION_AFTER.to_string(),
        _ => PASSWORD_DIRECTION_BEFORE.to_string(),
    }
}

    fn intercept_and_decipher_control(
        text: &str,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        len: usize,
        password_direction: &str,
        control_symbols: &str,
        cipher: &Self,
    ) -> Option<String> {
        if len == 0 {
            return None;
        }
        let chars: Vec<char> = text.chars().collect();
        let plen = len.max(2);
        let segment =
            Self::extract_control_segment(&chars, plen, password_direction, control_symbols)?;
        let deciphered = cipher.ocultar_palabra_with_control_symbols(
            &segment,
            -caja1,
            -caja2,
            -caja3,
            -caja4,
            caja5,
            "descifrar",
            control_symbols,
        );
        Some(deciphered)
    }

    fn extract_control_segment(
        chars: &[char],
        plen: usize,
        password_direction: &str,
        control_symbols: &str,
    ) -> Option<String> {
        for (index, char_value) in chars.iter().enumerate() {
            if !control_symbols.contains(*char_value) {
                continue;
            }

            if password_direction == PASSWORD_DIRECTION_AFTER {
                if index + 1 >= plen {
                    return Some(chars[index + 1 - plen..index + 1].iter().collect());
                }
            } else if index + plen <= chars.len() {
                return Some(chars[index..index + plen].iter().collect());
            }
        }

        None
    }

    fn ocultar_palabra_internal(
        &self,
        palabra: &str,
        caja1: i64,
        caja2: i64,
        caja3: i64,
        caja4: i64,
        caja5: i64,
        modo: &str,
        preserve_case: bool,
        preserve_control_symbols: bool,
        control_symbols: &str,
    ) -> String {
        let mut resultado = String::with_capacity(palabra.len());
        let espacio_en_plantilla = self.template.space_in_template(&self.grid);
        let total_chars = palabra.chars().count();
        let indice = total_chars as i64;

        for (i, caracter) in palabra.chars().enumerate() {
            if preserve_control_symbols && control_symbols.contains(caracter) {
                resultado.push(caracter);
                continue;
            }

            let caracter_busqueda = if caracter == ' ' { '␣' } else { caracter };
            let pos = self.template.find_position(&self.grid, caracter_busqueda);

            if let Some((fila, columna)) = pos {
                let mut c = columna as i64;
                let mut f = fila as i64;

                c += caja1 - indice;
                f += caja2 - indice;
                c -= caja3 - indice;
                f -= caja4 - indice;

                let nr = f.rem_euclid(self.template.rows as i64) as usize;
                let nc = c.rem_euclid(self.template.cols as i64) as usize;

                // ── caja5 — idéntico a Python 1.1 obtener_movimientos_caja5 ──
                let mut fr = nr as i64;
                let mut fc = nc as i64;
                for (mf, mc) in caja5_movements(i as i64, caja5, total_chars) {
                    if modo == "cifrar" {
                        fr += mf;
                        fc += mc;
                    } else {
                        fr -= mf;
                        fc -= mc;
                    }
                }
                let nr_final = fr.rem_euclid(self.template.rows as i64) as usize;
                let nc_final = fc.rem_euclid(self.template.cols as i64) as usize;

                let mut nuevo_caracter = self.grid[nr_final][nc_final];
                if modo == "descifrar" && nuevo_caracter == '␣' {
                    nuevo_caracter = ' ';
                }
                if preserve_case && caracter.is_uppercase() {
                    nuevo_caracter = nuevo_caracter
                        .to_uppercase()
                        .next()
                        .unwrap_or(nuevo_caracter);
                }
                resultado.push(nuevo_caracter);
            } else if caracter == ' ' && !espacio_en_plantilla {
                resultado.push(' ');
            } else {
                resultado.push(caracter);
            }
        }

        resultado
    }
}

/// caja5 2.0: patrón L original (1.1) en impares + espejo L⁻¹ en pares.
static CAJA5_PATTERNS: [[(i64, i64); 2]; 8] = [
    [(2, 0), (0, 1)],   // 0: impares, caja5%4=0 (↓↓→)
    [(-2, 0), (0, -1)], // 1: pares,   caja5%4=0 espejo (↑↑←)
    [(0, 2), (1, 0)],   // 2: impares, caja5%4=1 (→→↓)
    [(0, -2), (-1, 0)], // 3: pares,   caja5%4=1 espejo (←←↑)
    [(-2, 0), (0, -1)], // 4: impares, caja5%4=2 (↑↑←)
    [(2, 0), (0, 1)],   // 5: pares,   caja5%4=2 espejo (↓↓→)
    [(0, -2), (-1, 0)], // 6: impares, caja5%4=3 (←←↑)
    [(0, 2), (1, 0)],   // 7: pares,   caja5%4=3 espejo (→→↓)
];

/// Centinela: caja5 == total_chars-1 desactiva completamente.
#[inline(always)]
fn caja5_movements(i: i64, caja5: i64, total_chars: usize) -> &'static [(i64, i64); 2] {
    if caja5 == (total_chars as i64) - 1 {
        return &[(0, 0), (0, 0)]; // sin efecto (el for suma 0)
    }
    let base = (caja5.rem_euclid(4) * 2) as usize;
    let idx = if i % 2 == 1 { base } else { base + 1 }; // impar→original, par→espejo
    &CAJA5_PATTERNS[idx]
}
